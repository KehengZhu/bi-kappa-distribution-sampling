5class GeneralVelocityGenerator:
6 """Samples velocities from a user-defined speed-squared distribution g(w), w = |v|^2.
7
8 The speed is |v| = sqrt(w); the direction is isotropic in 3D. (For a
9 distribution originally posed in energy, w = |v|^2 = 2E/m, so the mass drops
10 out of this generator entirely.)"""
11
12 def __init__(self, speed_sq_pdf: Callable, speed_sq_min: float, speed_sq_max: float,
13 probe_points: int = 1024, max_reject_tries: int = 100000):
14 if speed_sq_min < 0.0:
15 raise ValueError("speed_sq_min must be non-negative (w = |v|^2 >= 0)")
16 if not (speed_sq_min < speed_sq_max):
17 raise ValueError("speed_sq_min must be strictly less than speed_sq_max")
18 self.speed_sq_pdf = speed_sq_pdf
19 self.speed_sq_min = speed_sq_min
20 self.speed_sq_max = speed_sq_max
21 self.max_reject_tries = max_reject_tries
22
23 # Estimate upper bound of the PDF
24 self.pdf_upper_bound = self._estimate_pdf_upper_bound(probe_points)
25
26 def _estimate_pdf_upper_bound(self, probe_points: int) -> float:
27 """Estimate the maximum value of the speed-squared PDF in the given range."""
28 grid = np.linspace(self.speed_sq_min, self.speed_sq_max, probe_points)
29 pdf_values = np.array([self.speed_sq_pdf(w) for w in grid])
30 return np.max(pdf_values)
31
32 def sample_speed_sq(self, rng: np.random.Generator) -> float:
33 """Sample w = |v|^2 using rejection sampling."""
34 for _ in range(self.max_reject_tries):
35 w = rng.uniform(self.speed_sq_min, self.speed_sq_max)
36 u = rng.uniform(0, self.pdf_upper_bound)
37 if u <= self.speed_sq_pdf(w):
38 return w
39 raise RuntimeError("Failed to sample speed_sq after max_reject_tries attempts")
40
41 def __call__(self, rng: np.random.Generator) -> np.ndarray:
42 """Generate a random 3D velocity vector."""
43 speed = np.sqrt(self.sample_speed_sq(rng))
44
45 # Isotropic direction on sphere
46 cos_theta = rng.uniform(-1.0, 1.0)
47 sin_theta = np.sqrt(max(0.0, 1.0 - cos_theta**2))
48 phi = rng.uniform(0.0, 2.0 * np.pi)
49
50 vx = speed * sin_theta * np.cos(phi)
51 vy = speed * sin_theta * np.sin(phi)
52 vz = speed * cos_theta
53
54 return np.array([vx, vy, vz])

57class GeneralPositionGenerator:
58 """Samples positions in 1D/2D/3D from a user density function using rejection sampling."""
59
60 def __init__(self, dimension: int, lower_bounds: List[float], upper_bounds: List[float],
61 density_function: Callable, probe_points: int = 4096, max_reject_tries: int = 200000):
62 self.dimension = dimension
63 self.lower_bounds = np.array(lower_bounds)
64 self.upper_bounds = np.array(upper_bounds)
65 self.density_function = density_function
66 self.max_reject_tries = max_reject_tries
67
68 # Estimate upper bound of the density
69 self.density_upper_bound = self._estimate_density_upper_bound(probe_points)
70
71 def _estimate_density_upper_bound(self, probe_points: int) -> float:
72 """Estimate the maximum value of the density function in the domain."""
73 max_density = 0.0
74 for _ in range(probe_points):
75 x = self.lower_bounds + np.random.rand(self.dimension) * (self.upper_bounds - self.lower_bounds)
76 rho = self.density_function(x.tolist())
77 max_density = max(max_density, rho)
78 return max_density
79
80 def __call__(self, rng: np.random.Generator) -> np.ndarray:
81 """Generate a random position vector using rejection sampling."""
82 for _ in range(self.max_reject_tries):
83 # Sample uniformly in the domain
84 x = self.lower_bounds + rng.uniform(0, 1, self.dimension) * (self.upper_bounds - self.lower_bounds)
85 u = rng.uniform(0, self.density_upper_bound)
86
87 if u <= self.density_function(x.tolist()):
88 # Pad to 3D if necessary
89 if self.dimension < 3:
90 x = np.pad(x, (0, 3 - self.dimension), mode='constant', constant_values=0)
91 return x
92
93 raise RuntimeError("Failed to sample position after max_reject_tries attempts")

96class FieldAlignedVelocityGenerator:
97 """Samples field-aligned velocities: an arbitrary parallel speed-squared
98 distribution g(w_par), w_par = v_par^2, along B, with a Maxwellian
99 perpendicular spread (thermal speed theta_perp). (For a distribution
100 originally posed in energy, w_par = v_par^2 = 2E_par/m, so the mass drops
101 out.)"""
102
103 def __init__(self, speed_sq_pdf: Callable, speed_sq_min: float, speed_sq_max: float,
104 theta_perp: float, ub: List[float], parallel_sign: int,
105 probe_points: int = 1024, max_reject_tries: int = 100000):
106 if speed_sq_min < 0.0:
107 raise ValueError("speed_sq_min must be non-negative (w = v_par^2 >= 0)")
108 if not (speed_sq_min < speed_sq_max):
109 raise ValueError("speed_sq_min must be strictly less than speed_sq_max")
110 if theta_perp <= 0.0:
111 raise ValueError("theta_perp must be positive")
112 if parallel_sign not in (1, -1):
113 raise ValueError("parallel_sign must be exactly +1 or -1")
114 ub = np.asarray(ub, dtype=float)
115 if np.dot(ub, ub) == 0.0:
116 raise ValueError("ub (magnetic field) must be a non-zero vector")
117
118 self.speed_sq_pdf = speed_sq_pdf
119 self.speed_sq_min = speed_sq_min
120 self.speed_sq_max = speed_sq_max
121 self.theta_perp = theta_perp
122 self.parallel_sign = float(parallel_sign)
123 self.max_reject_tries = max_reject_tries
124
125 # Field-aligned orthonormal frame (e1, e2 perpendicular; e3 = B/|B|).
126 self.e1, self.e2, self.e3 = self._build_frame(ub)
127
128 # Estimate upper bound of the parallel speed-squared PDF.
129 self.pdf_upper_bound = self._estimate_pdf_upper_bound(probe_points)
130
131 @staticmethod
132 def _build_frame(ub: np.ndarray):
133 """Right-handed orthonormal frame with e3 aligned to ub."""
134 e3 = ub / np.linalg.norm(ub)
135 # Pick a seed vector least aligned with e3 to stay well-conditioned.
136 seed = np.zeros(3)
137 seed[np.argmin(np.abs(e3))] = 1.0
138 e1 = np.cross(e3, seed)
139 e1 /= np.linalg.norm(e1)
140 e2 = np.cross(e3, e1)
141 return e1, e2, e3
142
143 def _estimate_pdf_upper_bound(self, probe_points: int) -> float:
144 grid = np.linspace(self.speed_sq_min, self.speed_sq_max, probe_points)
145 pdf_values = np.array([self.speed_sq_pdf(w) for w in grid])
146 return np.max(pdf_values) * 1.05
147
148 def sample_speed_sq(self, rng: np.random.Generator) -> float:
149 """Sample w_par = v_par^2 using rejection sampling."""
150 for _ in range(self.max_reject_tries):
151 w = rng.uniform(self.speed_sq_min, self.speed_sq_max)
152 u = rng.uniform(0, self.pdf_upper_bound)
153 if u <= self.speed_sq_pdf(w):
154 return w
155 raise RuntimeError("Failed to sample speed_sq after max_reject_tries attempts")
156
157 def __call__(self, rng: np.random.Generator) -> np.ndarray:
158 """Generate a random 3D velocity vector in the global frame."""
159 v_par = self.parallel_sign * np.sqrt(self.sample_speed_sq(rng))
160
161 # Perpendicular 2D Maxwellian: N(0, theta_perp^2/2) per component.
162 sigma_perp = self.theta_perp / np.sqrt(2.0)
163 v_perp1 = sigma_perp * rng.standard_normal()
164 v_perp2 = sigma_perp * rng.standard_normal()
165
166 return v_perp1 * self.e1 + v_perp2 * self.e2 + v_par * self.e3
