5class GeneralVelocityGenerator:
6 """Samples velocities from a user-defined energy distribution f(E)."""
7
8 def __init__(self, energy_pdf: Callable, energy_min: float, energy_max: float,
9 particle_mass: float, probe_points: int = 1024, max_reject_tries: int = 100000):
10 self.energy_pdf = energy_pdf
11 self.energy_min = energy_min
12 self.energy_max = energy_max
13 self.particle_mass = particle_mass
14 self.max_reject_tries = max_reject_tries
15
16 # Estimate upper bound of the PDF
17 self.pdf_upper_bound = self._estimate_pdf_upper_bound(probe_points)
18
19 def _estimate_pdf_upper_bound(self, probe_points: int) -> float:
20 """Estimate the maximum value of the energy PDF in the given range."""
21 energies = np.linspace(self.energy_min, self.energy_max, probe_points)
22 pdf_values = np.array([self.energy_pdf(E) for E in energies])
23 return np.max(pdf_values)
24
25 def sample_energy(self, rng: np.random.Generator) -> float:
26 """Sample energy using rejection sampling."""
27 for _ in range(self.max_reject_tries):
28 E = rng.uniform(self.energy_min, self.energy_max)
29 u = rng.uniform(0, self.pdf_upper_bound)
30 if u <= self.energy_pdf(E):
31 return E
32 raise RuntimeError("Failed to sample energy after max_reject_tries attempts")
33
34 def __call__(self, rng: np.random.Generator) -> np.ndarray:
35 """Generate a random 3D velocity vector."""
36 energy = self.sample_energy(rng)
37 speed = np.sqrt(2.0 * energy / self.particle_mass)
38
39 # Isotropic direction on sphere
40 cos_theta = rng.uniform(-1.0, 1.0)
41 sin_theta = np.sqrt(max(0.0, 1.0 - cos_theta**2))
42 phi = rng.uniform(0.0, 2.0 * np.pi)
43
44 vx = speed * sin_theta * np.cos(phi)
45 vy = speed * sin_theta * np.sin(phi)
46 vz = speed * cos_theta
47
48 return np.array([vx, vy, vz])

52 """Samples positions in 1D/2D/3D from a user density function using rejection sampling."""

54 def __init__(self, dimension: int, lower_bounds: List[float], upper_bounds: List[float],
55 density_function: Callable, probe_points: int = 4096, max_reject_tries: int = 200000):
56 self.dimension = dimension
57 self.lower_bounds = np.array(lower_bounds)
58 self.upper_bounds = np.array(upper_bounds)
59 self.density_function = density_function
60 self.max_reject_tries = max_reject_tries
61
62 # Estimate upper bound of the density
63 self.density_upper_bound = self._estimate_density_upper_bound(probe_points)
64
65 def _estimate_density_upper_bound(self, probe_points: int) -> float:
66 """Estimate the maximum value of the density function in the domain."""
67 max_density = 0.0
68 for _ in range(probe_points):
69 x = self.lower_bounds + np.random.rand(self.dimension) * (self.upper_bounds - self.lower_bounds)
70 rho = self.density_function(x.tolist())
71 max_density = max(max_density, rho)
72 return max_density
73
74 def __call__(self, rng: np.random.Generator) -> np.ndarray:
75 """Generate a random position vector using rejection sampling."""
76 for _ in range(self.max_reject_tries):
77 # Sample uniformly in the domain
78 x = self.lower_bounds + rng.uniform(0, 1, self.dimension) * (self.upper_bounds - self.lower_bounds)
79 u = rng.uniform(0, self.density_upper_bound)
80
81 if u <= self.density_function(x.tolist()):
82 # Pad to 3D if necessary
83 if self.dimension < 3:
84 x = np.pad(x, (0, 3 - self.dimension), mode='constant', constant_values=0)
85 return x
86
87 raise RuntimeError("Failed to sample position after max_reject_tries attempts")
