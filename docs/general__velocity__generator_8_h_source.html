119 template <typename GeneratorType> point_type operator()(GeneratorType &gen) const
120 {
121 if (!m_isDefined)
122 {
123 throw std::runtime_error(
124 "general_velocity_generator must be define(...)d before sampling");
125 }
126
127 const RealType kOne= RealType(1.0);
128 const RealType kTwo= RealType(2.0);
129
130 RealType energy= sampleEnergy(gen);
131 RealType speed= std::sqrt(kTwo * energy / m_particleMass);
132
133 RealType cosTheta= m_distCosTheta(gen);
134 RealType sinTheta= std::sqrt(std::max(RealType(0.0), kOne - cosTheta * cosTheta));
135 RealType phi= m_distPhi(gen);
136
137 RealType vx= speed * sinTheta * std::cos(phi);
138 RealType vy= speed * sinTheta * std::sin(phi);
139 RealType vz= speed * cosTheta;
140
141 point_type point;
142 point[0]= vx;
143 point[1]= vy;
144 point[2]= vz;
145 return point;
146 }

174 RealType estimatePdfUpperBound(int probe_points) const
175 {
176 RealType max_value= RealType(0.0);
177 int i= 0;
178 for (i= 0; i < probe_points; ++i)

181 RealType energy= m_energyMin + ratio * (m_energyMax - m_energyMin);

193 template <typename GeneratorType> RealType sampleEnergy(GeneratorType &gen) const
194 {
195 int tries= 0;
196 for (tries= 0; tries < m_maxRejectTries; ++tries)
197 {
198 RealType energy= m_distEnergy(gen);
199 RealType y= m_distAccept(gen);
200 RealType pdf_value= evaluateEnergyPdf(energy);
201 if (y <= pdf_value)
202 {
203 return energy;
204 }
205 }
206
207 throw std::runtime_error("failed to sample energy from energy_pdf: rejection sampling "
208 "exceeded max_reject_tries");
