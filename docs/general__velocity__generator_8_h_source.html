120 template <typename GeneratorType> point_type operator()(GeneratorType &gen)
121 {
122 if (!m_isDefined)
123 {
124 throw std::runtime_error(
125 "general_velocity_generator must be define(...)d before sampling");
126 }
127
128 const RealType kOne= RealType(1.0);
129
130 RealType speed= std::sqrt(sampleSpeedSq(gen)); // |V| = sqrt(w), w = |V|^2
131
132 RealType cosTheta= m_distCosTheta(gen);
133 RealType sinTheta= std::sqrt(std::max(RealType(0.0), kOne - cosTheta * cosTheta));
134 RealType phi= m_distPhi(gen);
135
136 RealType vx= speed * sinTheta * std::cos(phi);
137 RealType vy= speed * sinTheta * std::sin(phi);
138 RealType vz= speed * cosTheta;
139
140 point_type point;
141 point[0]= vx;
142 point[1]= vy;
143 point[2]= vz;
