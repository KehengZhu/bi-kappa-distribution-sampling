205 template <typename GeneratorType> point_type operator()(GeneratorType &gen)
206 {
207 if (!m_isDefined)
208 {
209 throw std::runtime_error(
210 "field_aligned_velocity_generator must be define(...)d before sampling");
211 }
212
213 const RealType sigmaScale= RealType(1.0) / std::sqrt(RealType(2.0));
214
215 // Parallel: v_par^2 -> signed speed along B.
216 RealType vPar= m_parallelSign * std::sqrt(sampleSpeedSq(gen));
217
218 // Perpendicular: 2D Maxwellian, N(0, theta_perp^2/2) per component, with the
219 // pair redrawn while either component exceeds the cap.
220 RealType vPerp1= RealType(0.0), vPerp2= RealType(0.0);
221 const RealType perpMax= m_maxNormalizedVelocity * m_thetaPerp;
222 int tries= 0;
223 for (tries= 0; tries < m_maxRejectTries; ++tries)
224 {
225 vPerp1= m_thetaPerp * sigmaScale * m_standardNormal(gen);
226 vPerp2= m_thetaPerp * sigmaScale * m_standardNormal(gen);
227 if (std::abs(vPerp1) <= perpMax && std::abs(vPerp2) <= perpMax)
228 break;
229 }
230 if (tries == m_maxRejectTries)
231 {
232 throw std::runtime_error("failed to sample the perpendicular components under "
233 "max_normalized_velocity: exceeded max_reject_tries");
234 }
235
236 // Field-aligned frame: (perp1 -> e1, perp2 -> e2, par -> e3 = B/|B|).
237 point_type local;
238 local[0]= vPerp1;
239 local[1]= vPerp2;
240 local[2]= vPar;
241 return rotate_from_fieldAligned_frame(local, m_ub);
242 }

258 // Geometry helper
259 // ====================================================================
260 // Transform local velocity components (perp1, perp2, par) to global
261 // Cartesian coordinates given the parallel direction ub.
262 point_type rotate_from_fieldAligned_frame(const point_type &local, const point_type &ub) const
263 {
264 const RealType kZero= RealType(0.0);
265 const RealType kOne= RealType(1.0);
266
267 RealType ubNorm2= ub[0] * ub[0] + ub[1] * ub[1] + ub[2] * ub[2];
268 if (approximatelyEqual(ubNorm2, kZero))
269 {
270 throw std::invalid_argument("ub must be a non-zero vector");
271 }
272
273 RealType ubNorm= std::sqrt(ubNorm2);
274
275 point_type e3;
276 e3[0]= ub[0] / ubNorm;
277 e3[1]= ub[1] / ubNorm;
278 e3[2]= ub[2] / ubNorm;
279
280 point_type e2({kOne, kOne, kOne});
281
282 int maxcomp= 0;
283 if (std::abs(e3[1]) > std::abs(e3[maxcomp]))
284 {
285 maxcomp= 1;
286 }
287 if (std::abs(e3[2]) > std::abs(e3[maxcomp]))
288 {
289 maxcomp= 2;
290 }
291
292 RealType sum= e3[0] + e3[1] + e3[2];
293 e2[maxcomp]= kOne - sum / e3[maxcomp];
294
295 RealType e2Norm2= e2[0] * e2[0] + e2[1] * e2[1] + e2[2] * e2[2];
296 RealType e2Norm= std::sqrt(e2Norm2);
297 e2[0]/= e2Norm;
298 e2[1]/= e2Norm;
299 e2[2]/= e2Norm;
300
301 point_type e1;
302 e1[0]= e2[1] * e3[2] - e2[2] * e3[1];
303 e1[1]= e2[2] * e3[0] - e2[0] * e3[2];
304 e1[2]= e2[0] * e3[1] - e2[1] * e3[0];
305
306 point_type point;
307 point[0]= local[0] * e1[0] + local[1] * e2[0] + local[2] * e3[0];
308 point[1]= local[0] * e1[1] + local[1] * e2[1] + local[2] * e3[1];
309 point[2]= local[0] * e1[2] + local[1] * e2[2] + local[2] * e3[2];
310 return point;
311 }

329 RealType estimatePdfUpperBound(int probe_points) const
330 {
331 RealType max_value= RealType(0.0);
332 int i= 0;
333 for (i= 0; i < probe_points; ++i)
334 {
335 RealType ratio= RealType(i) / RealType(probe_points - 1);
336 RealType speed_sq= m_speedSqMin + ratio * (m_speedSqMax - m_speedSqMin);
337 RealType value= evaluateSpeedSqPdf(speed_sq);
338 if (value > max_value)
339 {
340 max_value= value;
341 }
342 }
343
344 // Add safety margin in case peak lies between probes.
345 return max_value * RealType(1.05);

348 template <typename GeneratorType> RealType sampleSpeedSq(GeneratorType &gen)
349 {
350 if (!(m_speedSqMin < m_speedSqMax))
351 {
352 return m_speedSqMin;
353 }
354 int tries= 0;
355 for (tries= 0; tries < m_maxRejectTries; ++tries)
356 {
357 RealType speed_sq= m_distSpeedSq(gen);
358 RealType y= m_distAccept(gen);
359 RealType pdf_value= evaluateSpeedSqPdf(speed_sq);
360 if (y <= pdf_value)
361 {
362 return speed_sq;
363 }
364 }
365
366 throw std::runtime_error("failed to sample speed_sq from speed_sq_pdf: rejection sampling "
367 "exceeded max_reject_tries");
368 }
