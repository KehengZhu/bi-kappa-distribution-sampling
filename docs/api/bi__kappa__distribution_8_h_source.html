19// This sampler decides what it may return by materializing a value and asking whether it is
20// finite, and no_cap() is +infinity. `-ffast-math` (which defines __FAST_MATH__) licenses a
21// compiler to assume that no infinity or NaN occurs, and so to fold `mag <= max()` to `true`:
22// an overflowed component could then be returned as if it were a number, n_nonfinite() would
23// read zero, and no_cap() would not be reliable. A finite cap (the default of 20) still
24// rejects such draws, because it compares against a finite bound. The header therefore
25// compiles under fast-math but says so; define BI_KAPPA_ALLOW_FAST_MATH to silence the note.
26#if defined(__FAST_MATH__) && !defined(BI_KAPPA_ALLOW_FAST_MATH)
27#pragma message("bi_kappa_distribution.H: compiled with -ffast-math; no_cap() and the overflow " \
28 "counters are unreliable in this mode. Define BI_KAPPA_ALLOW_FAST_MATH to " \
29 "silence this note.")
30#endif

56template <typename GeneratorType> inline int engineBits()
57{
58 unsigned long long range= static_cast<unsigned long long>(GeneratorType::max()) -
59 static_cast<unsigned long long>(GeneratorType::min());
60 int bits= 0;
61 while (range >= 1ULL)
62 {
63 ++bits;
64 range>>= 1;
65 }
66 return bits;
67}

83template <typename RealType, typename GeneratorType> RealType openCanonical(GeneratorType &gen)
84{
85 const int bits= std::numeric_limits<RealType>::digits - 1;
86 const int per= engineBits<GeneratorType>();
87 unsigned long long k= 0;
88 int have= 0;
89 while (have < bits)
90 {
91 const unsigned long long word= static_cast<unsigned long long>(gen()) -
92 static_cast<unsigned long long>(GeneratorType::min());
93 const int take= (bits - have < per) ? (bits - have) : per;
94 k= (k << take) | (word & ((1ULL << take) - 1ULL));
95 have+= take;
96 }
97 return (static_cast<RealType>(k) + RealType(0.5)) * std::ldexp(RealType(1.0), -bits);
98}

103template <typename RealType, typename GeneratorType>
104RealType polarNormal(GeneratorType &gen, RealType &cache, bool &cached)
105{
106 if (cached)
107 {
108 cached= false;
109 return cache;
110 }
111 RealType x, y, s;
112 do
113 {
114 x= RealType(2.0) * openCanonical<RealType, GeneratorType>(gen) - RealType(1.0);
115 y= RealType(2.0) * openCanonical<RealType, GeneratorType>(gen) - RealType(1.0);
116 s= x * x + y * y;
117 } while (s >= RealType(1.0));
118 // s == 0 is unreachable: openCanonical never returns exactly 1/2, so neither x nor y is
119 // ever exactly zero.
120 const RealType f= std::sqrt(RealType(-2.0) * std::log(s) / s);
121 cache= y * f;
122 cached= true;
123 return x * f;
124}

129template <typename RealType, typename GeneratorType>
130RealType marsagliaTsangGamma(RealType shape, GeneratorType &gen, RealType &ncache, bool &ncached)
131{
132 const RealType d= shape - RealType(1.0) / RealType(3.0);
133 const RealType c= RealType(1.0) / std::sqrt(RealType(9.0) * d);
134 for (;;)
135 {
136 RealType x, v;
137 do
138 {
139 x= polarNormal<RealType, GeneratorType>(gen, ncache, ncached);
140 v= RealType(1.0) + c * x;
141 } while (v <= RealType(0.0));
142 v= v * v * v;
143 const RealType u= openCanonical<RealType, GeneratorType>(gen);
144 const RealType xx= x * x;
145 if (u < RealType(1.0) - RealType(0.0331) * xx * xx)
146 {
147 return d * v;
148 }
149 if (std::log(u) < RealType(0.5) * xx + d * (RealType(1.0) - v + std::log(v)))
150 {
151 return d * v;
152 }
153 }
154}

183template <typename RealType, typename GeneratorType>
184RealType logGamma(RealType a, GeneratorType &gen, RealType &ncache, bool &ncached)
185{
186 const RealType y=
187 marsagliaTsangGamma<RealType, GeneratorType>(a + RealType(1.0), gen, ncache, ncached);
188 const RealType u= openCanonical<RealType, GeneratorType>(gen);
189 return std::log(y) + std::log(u) / a;
190}

230template <typename RealType>
231bool materializeComponents(RealType logR, const std::array<RealType, 3> &g,
232 std::array<RealType, 3> &point)
233{
234 const RealType maxFinite= std::numeric_limits<RealType>::max();
235 const RealType rTry= std::exp(logR);
236 // `false` for +infinity and for NaN, both of which fail every comparison.
237 const bool radiusRepresentable= (rTry <= maxFinite);
238 const RealType R= radiusRepresentable ? rTry : RealType(0.0);
239
240 bool anyOverflow= false;
241 for (int j= 0; j < 3; ++j)
242 {
243 if (g[j] == RealType(0.0))
244 {
245 point[j]= RealType(0.0);
246 continue;
247 }
248 const RealType absG= std::abs(g[j]);
249 // What decides representability is the number the caller will receive. A
250 // magnitude that rounds past `max()` becomes an infinity here and is counted;
251 // deciding from a logarithm instead would be wrong in both directions at once.
252 const RealType mag= radiusRepresentable ? R * absG
253 : std::exp(logR + std::log(absG));
254 if (!(mag <= maxFinite))
255 {
256 anyOverflow= true;
257 point[j]= (g[j] < RealType(0.0)) ? -std::numeric_limits<RealType>::infinity()
258 : std::numeric_limits<RealType>::infinity();
259 continue;
260 }
261 point[j]= (g[j] < RealType(0.0)) ? -mag : mag;
262 }
263 return anyOverflow;
264}

331 param_type(RealType kappa= RealType(2.0), RealType theta_perp= RealType(1.0),
332 RealType theta_par= RealType(1.0),
333 point_type ub= point_type({RealType(0.0), RealType(0.0), RealType(1.0)}),
334 RealType max_normalized_velocity= RealType(20.0))
335 : m_kappa(kappa), m_theta_perp(theta_perp), m_theta_par(theta_par), m_ub(ub),
336 m_maxNormalizedVelocity(max_normalized_velocity)
337 {
338 // Each test is written so that NaN fails it. `x <= 0.5` is false for a NaN,
339 // so the natural spelling admits one, and a NaN parameter then propagates into
340 // log R, where it can no longer be told from an unrepresentable velocity: the
341 // sampler would report a non-finite draw and count it as an overflow, which it
342 // is not. Rejecting the parameter is the only point at which the two are
343 // still distinguishable.
344 if (!(m_kappa > RealType(0.5)))
345 {
346 throw std::invalid_argument("kappa must be > 0.5 for normalization");
347 }
348 if (!(m_theta_perp > RealType(0.0)) || !(m_theta_par > RealType(0.0)) ||
349 !(m_theta_perp <= std::numeric_limits<RealType>::max()) ||
350 !(m_theta_par <= std::numeric_limits<RealType>::max()))
351 {
352 throw std::invalid_argument(
353 "Temperature parameters must be positive and finite");
354 }
355 {
356 // The field direction is validated here rather than inside the rotation, so
357 // that an axis-aligned-looking NaN cannot slip past the fast path.
358 RealType n2= RealType(0.0);
359 for (int j= 0; j < 3; ++j)
360 {
361 if (!(std::abs(m_ub[j]) <= std::numeric_limits<RealType>::max()))
362 {
363 throw std::invalid_argument("ub components must be finite");
364 }
365 n2+= m_ub[j] * m_ub[j];
366 }
367 if (!(n2 > RealType(0.0)))
368 {
369 throw std::invalid_argument("ub must be a non-zero vector");
370 }
371 }
372 // A positive finite value selects the capped mode. no_cap() (= +inf)
373 // selects the untruncated mode, in which no acceptance-rejection loop
374 // runs at all and samples follow the full bi-Kappa law.
375 if (!(m_maxNormalizedVelocity > RealType(0.0)))
376 {
377 throw std::invalid_argument(
378 "max_normalized_velocity must be positive, or no_cap() for the "
379 "untruncated distribution");
380 }
381 }

673 template <typename GeneratorType>
674 result_type operator()(GeneratorType &gen, const param_type &params)
675 {
676 const RealType kHalf= RealType(0.5);
677 const RealType kOne= RealType(1.0);
678 const int kMaxCapRejectTries= 1000000;
679
680 RealType kappa= params.kappa();
681 RealType theta_perp= params.theta_perp();
682 RealType theta_par= params.theta_par();
683 point_type ub= params.ub();
684 RealType maxNormalizedVelocity= params.max_normalized_velocity();
685
686 const RealType a= kappa - kHalf;
687 const RealType sqrtKappa= std::sqrt(kappa);
688
689 // Untruncated mode runs the core mapping exactly once; capped mode redraws
690 // until the sample falls inside the box.
691 const bool capped= params.capped();
692 const int maxTries= capped ? kMaxCapRejectTries : 1;
693
694 const bool axisAligned= approximatelyEqual(ub[0], RealType(0.0)) &&
695 approximatelyEqual(ub[1], RealType(0.0)) &&
696 approximatelyEqual(ub[2], RealType(1.0));
697
698 // Four logarithms that only the capped branch reads, and so are only taken when
699 // there is a cap. Untruncated mode decides representability from the components
700 // themselves and needs no logarithm at all outside step 1; computing these anyway
701 // cost two `std::log` calls on every untruncated draw, which `-fmath-errno` (the
702 // default) forbids a compiler from removing.
703 //
704 // `logMax` is a screening quantity only: the cap branch uses it to decide whether a
705 // rejected draw is near enough to the type's limit to be worth materializing.
706 // Representability itself is never decided from a logarithm -- see
707 // bikappa_detail::materializeComponents.
708 //
709 // `logGBound` bounds the *velocity* from its radius alone:
710 // |g_j| <= |g| <= sqrt(kappa) max(theta_perp, theta_par) for every direction and
711 // every rotation, because a rotation preserves the norm. That is what the capped
712 // branch needs and what the normalized coordinate the cap test is written in cannot
713 // give it. It is finite because define() rejects a non-positive or non-finite
714 // theta, so the screen below is never comparing against a NaN.
715 const RealType logMax= capped ? std::log(std::numeric_limits<RealType>::max())
716 : RealType(0.0);
717 const RealType logCap= capped ? std::log(maxNormalizedVelocity) : RealType(0.0);
718 const RealType logSqrtKappa= capped ? std::log(sqrtKappa) : RealType(0.0);
719 const RealType logGBound=
720 capped ? logSqrtKappa + std::log(std::max(theta_perp, theta_par)) : RealType(0.0);
721
722 int tries= 0;
723 for (tries= 0; tries < maxTries; ++tries)
724 {
725 ++m_attempts;
726
727 // Step 1: the radius, carried as its logarithm.
728 //
729 // R = sqrt(X1)/sqrt(X2) with X1 ~ Gamma(3/2,1) and X2 ~ Gamma(a,1), a = kappa-1/2.
730 // As kappa -> 1/2 the shape a -> 0 and X2 underflows to exactly zero, which no
731 // rearrangement of sqrt(X1)/sqrt(X2) can survive. X2 is therefore never formed:
732 // log X2 comes from the shape-boosting identity and
733 //
734 // log R = (log X1 - log X2) / 2
735 //
736 // stays finite and accurate wherever the mathematical radius exists. X1 has
737 // shape 3/2 > 1, so it is drawn directly and its logarithm is well conditioned.
738 const RealType x1=
739 bikappa_detail::marsagliaTsangGamma<RealType, GeneratorType>(
740 RealType(1.5), gen, m_normalCache, m_normalCached);
741 const RealType logX2=
742 bikappa_detail::logGamma<RealType, GeneratorType>(a, gen, m_normalCache,
743 m_normalCached);
744 const RealType logR= kHalf * (std::log(x1) - logX2);
745
746 // Step 2: a direction uniform on S^2, by the rejection method of Marsaglia
747 // (1972), Ann. Math. Statist. 43(2), 645-646. Draw a point uniformly in the
748 // unit disc and lift it; the lift is area-preserving, so the result is exactly
749 // uniform on the sphere.
750 //
751 // The textbook alternative -- cos(theta) uniform on (-1,1) and phi uniform on
752 // (0, 2pi) -- is equally correct as mathematics, and is what release 1.0.0
753 // used. It needs sin(phi) and cos(phi) of the same argument, and a
754 // compiler is free to fuse that pair into one combined routine. Clang does, on
755 // Darwin, into __sincos_stret, whose sine differs from the standalone sin by one
756 // unit in the last place for roughly one argument in a thousand; GCC calls the
757 // two separately. Two builds of this header then return different numbers for
758 // the same seed, and the difference belongs to the compiler rather than to the
759 // sampler. The rejection method uses only sqrt, which IEEE-754 requires to be
760 // correctly rounded, so the direction is identical under every compiler. It is
761 // also faster here, despite averaging 4/pi = 1.273 attempts per direction.
762 RealType d1, d2, dsq;
763 do
764 {
765 d1= RealType(2.0) *
766 bikappa_detail::openCanonical<RealType, GeneratorType>(gen) - kOne;
767 d2= RealType(2.0) *
768 bikappa_detail::openCanonical<RealType, GeneratorType>(gen) - kOne;
769 dsq= d1 * d1 + d2 * d2;
770 } while (dsq >= kOne);
771 // dsq == 0 is unreachable: openCanonical never returns exactly 1/2, so neither
772 // d1 nor d2 is ever exactly zero.
773 const RealType dscale= RealType(2.0) * std::sqrt(kOne - dsq);
774
775 const RealType nx= d1 * dscale;
776 const RealType ny= d2 * dscale;
777 const RealType nz= kOne - RealType(2.0) * dsq;
778
779 // Step 3: the cap, tested in the field-aligned frame before anything is
780 // exponentiated. This is the same predicate the linear form applies --
781 // |point_j| / theta_j = sqrt(kappa) R |n_j| <= lambda -- moved into the log
782 // domain so that an intended sample far outside the box is rejected rather
783 // than overflowed first and rejected afterwards. Only the largest component
784 // has to be examined, so this costs one logarithm rather than three.
785 if (capped)
786 {
787 const RealType maxAbsN=
788 std::max(std::abs(nx), std::max(std::abs(ny), std::abs(nz)));
789 const RealType logLargest= logR + logSqrtKappa + std::log(maxAbsN);
790 if (logLargest > logCap)
791 {
792 // An intended sample that is outside the box and also unrepresentable is
793 // still counted, so a caller can tell the two apart. Two things have to
794 // be right here.
795 //
796 // *What* is tested. Representability is a property of the velocity
797 // `R g_j`, not of the normalized coordinate `sqrt(kappa) R |n_j|` the cap
798 // test just used: the two differ by `theta` and by the rotation. The
799 // ratio is bounded by `sqrt(3) max(theta_perp, theta_par)/min(...)`,
800 // since `|w|/sqrt(3) <= max_j |w_j| <= |w|` -- at
801 // `theta_par/theta_perp = 2` that is 3.46. The rotation concentrates the
802 // largest component as readily as it spreads it, which is why the bound
803 // is not simply the theta ratio. The counter must answer the question
804 // n_nonfinite() documents, which is about the velocity.
805 //
806 // *How* it is tested. By materializing, for the same reason as step 5.
807 // The exponential is worth avoiding here -- in capped mode this is the
808 // hot branch -- so `logGBound` screens first. It is an upper bound on
809 // log|V_j| that holds for every direction and every rotation, so a draw a
810 // whole natural-log unit below the limit by that bound cannot overflow.
811 //
812 // The screen is sound, not merely generous. Its error is dominated by
813 // half an ulp of `logR + logGBound` at the limit, which is 88.72 * 2^-24
814 // = 5.3e-6 in `float` and 709.78 * 2^-53 = 7.9e-14 in `double`; with the
815 // libm and normalization terms the worst case is about 1.1e-5 and
816 // 1.0e-13. One natural-log unit exceeds those by 9.4e4 and 1.0e13.
817 //
818 // How often it fires is a property of the configuration, not a constant:
819 // about 7e-7 of attempts at double kappa = 0.51, but 17 % at float
820 // kappa = 0.51 and 87 % at double kappa = 0.5001, where most of the law
821 // genuinely lies outside the type. Where it fires often, it is doing
822 // the work it exists for.
823 if (logR + logGBound > logMax - kOne)
824 {
825 point_type gRej= fieldAlignedG(sqrtKappa, theta_perp, theta_par,
826 nx, ny, nz);
827 if (!axisAligned)
828 {
829 gRej= rotateFromFieldAlignedFrame(gRej, ub);
830 }
831 point_type scratch;
832 if (bikappa_detail::materializeComponents<RealType>(logR, gRej,
833 scratch))
834 {
835 ++m_nonfinite;
836 }
837 }
838 continue;
839 }
840 }
841
842 // Step 4: the order-unity vector g, with V = R g. Forming g first is what makes
843 // the representability test meaningful: an orthogonal rotation can spread one
844 // over-large local component across three representable global ones, so the test
845 // has to be applied to the components that are actually returned.
846 point_type g= fieldAlignedG(sqrtKappa, theta_perp, theta_par, nx, ny, nz);
847 if (!axisAligned)
848 {
849 g= rotateFromFieldAlignedFrame(g, ub);
850 }
851
852 // Step 5: materialize the components and decide representability from the
853 // components themselves, never from their logarithms. The whole of that
854 // decision lives in bikappa_detail::materializeComponents, which documents why
855 // a logarithmic test is wrong in both directions; the capped branch above calls
856 // the same routine, so the counter and the return can never disagree.
857 point_type point;
858 const bool anyOverflow=
859 bikappa_detail::materializeComponents<RealType>(logR, g, point);
860
861 if (anyOverflow)
862 {
863 // True overflow: the intended velocity has no representation in RealType.
864 // No formation can rescue it, so it is returned as a non-finite vector and
865 // counted, never silently redrawn. In capped mode this is unreachable for
866 // every cap a caller would write: an accepted draw has
867 // sqrt(kappa) R max_j|n_j| <= lambda, hence
868 // max_j |g_j| R <= sqrt(3) lambda max(theta_perp, theta_par), so reaching it
869 // would take lambda max(theta) above 1.97e38 in `float` or 1.04e308 in
870 // `double` -- a box larger than the type.
871 ++m_nonfinite;
872 }
873
874 return point;
875 }
876
877 throw std::runtime_error("failed to sample velocity under max_normalized_velocity cap: "
878 "rejection sampling exceeded internal attempt limit");
879 }

894 static point_type rotateFromFieldAlignedFrame(const point_type &local, const point_type &ub)
895 {
896 const RealType kZero= RealType(0.0);
897 const RealType kOne= RealType(1.0);
898
899 // Normalize ub to get e3
900 RealType ubNorm2= ub[0] * ub[0] + ub[1] * ub[1] + ub[2] * ub[2];
901 if (approximatelyEqual(ubNorm2, kZero))
902 {
903 throw std::invalid_argument("ub must be a non-zero vector");
904 }
905
906 RealType ubNorm= std::sqrt(ubNorm2);
907
908 point_type e3;
909 e3[0]= ub[0] / ubNorm;
910 e3[1]= ub[1] / ubNorm;
911 e3[2]= ub[2] / ubNorm;
912
913 // Build e2 perpendicular to e3
914 point_type e2;
915 e2[0]= kOne;
916 e2[1]= kOne;
917 e2[2]= kOne;
918
919 int maxcomp= 0;
920 if (std::abs(e3[1]) > std::abs(e3[maxcomp]))
921 {
922 maxcomp= 1;
923 }
924 if (std::abs(e3[2]) > std::abs(e3[maxcomp]))
925 {
926 maxcomp= 2;
927 }
928
929 RealType sum= e3[0] + e3[1] + e3[2];
930 e2[maxcomp]= kOne - sum / e3[maxcomp];
931
932 RealType e2Norm2= e2[0] * e2[0] + e2[1] * e2[1] + e2[2] * e2[2];
933 RealType e2Norm= std::sqrt(e2Norm2);
934 e2[0]/= e2Norm;
935 e2[1]/= e2Norm;
936 e2[2]/= e2Norm;
937
938 // e1 = e2 x e3
939 point_type e1;
940 e1[0]= e2[1] * e3[2] - e2[2] * e3[1];
941 e1[1]= e2[2] * e3[0] - e2[0] * e3[2];
942 e1[2]= e2[0] * e3[1] - e2[1] * e3[0];
943
944 // Transform local coordinates to Cartesian coordinates
945 point_type point;
946 point[0]= local[0] * e1[0] + local[1] * e2[0] + local[2] * e3[0];
947 point[1]= local[0] * e1[1] + local[1] * e2[1] + local[2] * e3[1];
948 point[2]= local[0] * e1[2] + local[1] * e2[2] + local[2] * e3[2];
949
950 return point;
951 }
