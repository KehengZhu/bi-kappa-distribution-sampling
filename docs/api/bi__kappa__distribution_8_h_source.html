43template <typename GeneratorType> inline int engineBits()
44{
45 unsigned long long range= static_cast<unsigned long long>(GeneratorType::max()) -
46 static_cast<unsigned long long>(GeneratorType::min());
47 int bits= 0;
48 while (range >= 1ULL)
49 {
50 ++bits;

52 }

70template <typename RealType, typename GeneratorType> RealType openCanonical(GeneratorType &gen)
71{
72 const int bits= std::numeric_limits<RealType>::digits - 1;
73 const int per= engineBits<GeneratorType>();
74 unsigned long long k= 0;
75 int have= 0;
76 while (have < bits)
77 {
78 const unsigned long long word= static_cast<unsigned long long>(gen()) -
79 static_cast<unsigned long long>(GeneratorType::min());
80 const int take= (bits - have < per) ? (bits - have) : per;
81 k= (k << take) | (word & ((1ULL << take) - 1ULL));
82 have+= take;
83 }
84 return (static_cast<RealType>(k) + RealType(0.5)) * std::ldexp(RealType(1.0), -bits);
85}

90template <typename RealType, typename GeneratorType>
91RealType polarNormal(GeneratorType &gen, RealType &cache, bool &cached)
92{
93 if (cached)
94 {
95 cached= false;
96 return cache;
97 }
98 RealType x, y, s;
99 do
100 {
101 x= RealType(2.0) * openCanonical<RealType, GeneratorType>(gen) - RealType(1.0);
102 y= RealType(2.0) * openCanonical<RealType, GeneratorType>(gen) - RealType(1.0);
103 s= x * x + y * y;
104 } while (s >= RealType(1.0));
105 // s == 0 is unreachable: openCanonical never returns exactly 1/2, so neither x nor y is
106 // ever exactly zero.
107 const RealType f= std::sqrt(RealType(-2.0) * std::log(s) / s);
108 cache= y * f;
109 cached= true;
110 return x * f;
111}

116template <typename RealType, typename GeneratorType>
117RealType marsagliaTsangGamma(RealType shape, GeneratorType &gen, RealType &ncache, bool &ncached)
118{
119 const RealType d= shape - RealType(1.0) / RealType(3.0);
120 const RealType c= RealType(1.0) / std::sqrt(RealType(9.0) * d);
121 for (;;)
122 {
123 RealType x, v;
124 do
125 {
126 x= polarNormal<RealType, GeneratorType>(gen, ncache, ncached);
127 v= RealType(1.0) + c * x;
128 } while (v <= RealType(0.0));
129 v= v * v * v;
130 const RealType u= openCanonical<RealType, GeneratorType>(gen);
131 const RealType xx= x * x;
132 if (u < RealType(1.0) - RealType(0.0331) * xx * xx)
133 {
134 return d * v;
135 }
136 if (std::log(u) < RealType(0.5) * xx + d * (RealType(1.0) - v + std::log(v)))
137 {
138 return d * v;
139 }
140 }
141}

166template <typename RealType, typename GeneratorType>
167RealType logGamma(RealType a, GeneratorType &gen, RealType &ncache, bool &ncached)
168{
169 const RealType y=
170 marsagliaTsangGamma<RealType, GeneratorType>(a + RealType(1.0), gen, ncache, ncached);

524 template <typename GeneratorType>
525 result_type operator()(GeneratorType &gen, const param_type &params)
526 {
527 const RealType kHalf= RealType(0.5);
528 const RealType kOne= RealType(1.0);
529 const int kMaxCapRejectTries= 1000000;
530
531 RealType kappa= params.kappa();
532 RealType theta_perp= params.theta_perp();
533 RealType theta_par= params.theta_par();
534 point_type ub= params.ub();
535 RealType maxNormalizedVelocity= params.max_normalized_velocity();
536
537 const RealType a= kappa - kHalf;
538 RealType sqrtKappa= std::sqrt(kappa);
539
540 // Untruncated mode runs the core mapping exactly once; capped mode redraws
541 // until the sample falls inside the box.
542 const bool capped= params.capped();
543 const int maxTries= capped ? kMaxCapRejectTries : 1;
544
545 const bool axisAligned= approximatelyEqual(ub[0], RealType(0.0)) &&
546 approximatelyEqual(ub[1], RealType(0.0)) &&
547 approximatelyEqual(ub[2], RealType(1.0));
548
549 // The largest finite magnitude, and its logarithm. A component is representable
550 // exactly when log|V_j| is below this; the comparison is made in the log domain,
551 // before anything is exponentiated.
552 const RealType logMax= std::log(std::numeric_limits<RealType>::max());
553 const RealType logCap= capped ? std::log(maxNormalizedVelocity)
554 : std::numeric_limits<RealType>::infinity();
555 const RealType logSqrtKappa= std::log(sqrtKappa);
556 // |g_j| never exceeds sqrt(kappa) max(theta_perp, theta_par), because a rotation
557 // preserves the norm; see step 5.
558 const RealType logGBound= logSqrtKappa + std::log(std::max(theta_perp, theta_par));
559
560 int tries= 0;
561 for (tries= 0; tries < maxTries; ++tries)
562 {
563 ++m_attempts;
564
565 // Step 1: the radius, carried as its logarithm.
566 //
567 // R = sqrt(X1)/sqrt(X2) with X1 ~ Gamma(3/2,1) and X2 ~ Gamma(a,1), a = kappa-1/2.
568 // As kappa -> 1/2 the shape a -> 0 and X2 underflows to exactly zero, which no
569 // rearrangement of sqrt(X1)/sqrt(X2) can survive. X2 is therefore never formed:
570 // log X2 comes from the shape-boosting identity and
571 //
572 // log R = (log X1 - log X2) / 2
573 //
574 // stays finite and accurate wherever the mathematical radius exists. X1 has
575 // shape 3/2 > 1, so it is drawn directly and its logarithm is well conditioned.
576 const RealType x1=
577 bikappa_detail::marsagliaTsangGamma<RealType, GeneratorType>(
578 RealType(1.5), gen, m_normalCache, m_normalCached);
579 const RealType logX2=
580 bikappa_detail::logGamma<RealType, GeneratorType>(a, gen, m_normalCache,
581 m_normalCached);
582 const RealType logR= kHalf * (std::log(x1) - logX2);
583
584 // Step 2: a direction uniform on S^2, by the rejection method of Marsaglia
585 // (1972), Ann. Math. Statist. 43(2), 645-646. Draw a point uniformly in the
586 // unit disc and lift it; the lift is area-preserving, so the result is exactly
587 // uniform on the sphere.
588 //
589 // The textbook alternative -- cos(theta) uniform on (-1,1) and phi uniform on
590 // (0, 2pi) -- is equally correct as mathematics, and is what this loader used
591 // before 2.0.0. It needs sin(phi) and cos(phi) of the same argument, and a
592 // compiler is free to fuse that pair into one combined routine. Clang does, on
593 // Darwin, into __sincos_stret, whose sine differs from the standalone sin by one
594 // unit in the last place for roughly one argument in a thousand; GCC calls the
595 // two separately. Two builds of this header then return different numbers for
596 // the same seed, and the difference belongs to the compiler rather than to the
597 // sampler. The rejection method uses only sqrt, which IEEE-754 requires to be
598 // correctly rounded, so the direction is identical under every compiler. It is
599 // also faster here, despite averaging 4/pi = 1.273 attempts per direction.
600 RealType d1, d2, dsq;
601 do
602 {
603 d1= RealType(2.0) *
604 bikappa_detail::openCanonical<RealType, GeneratorType>(gen) - kOne;
605 d2= RealType(2.0) *
606 bikappa_detail::openCanonical<RealType, GeneratorType>(gen) - kOne;
607 dsq= d1 * d1 + d2 * d2;
608 } while (dsq >= kOne);
609 // dsq == 0 is unreachable: openCanonical never returns exactly 1/2, so neither
610 // d1 nor d2 is ever exactly zero.
611 const RealType dscale= RealType(2.0) * std::sqrt(kOne - dsq);
612
613 const RealType nx= d1 * dscale;
614 const RealType ny= d2 * dscale;
615 const RealType nz= kOne - RealType(2.0) * dsq;
616
617 // Step 3: the cap, tested in the field-aligned frame before anything is
618 // exponentiated. This is the same predicate the linear form applies --
619 // |point_j| / theta_j = sqrt(kappa) R |n_j| <= lambda -- moved into the log
620 // domain so that an intended sample far outside the box is rejected rather
621 // than overflowed first and rejected afterwards. Only the largest component
622 // has to be examined, so this costs one logarithm rather than three.
623 if (capped)
624 {
625 const RealType maxAbsN=
626 std::max(std::abs(nx), std::max(std::abs(ny), std::abs(nz)));
627 const RealType logLargest= logR + logSqrtKappa + std::log(maxAbsN);
628 if (logLargest > logCap)
629 {
630 // An intended sample that is outside the box and also unrepresentable
631 // is still counted, so a caller can tell the two apart.
632 if (logLargest >= logMax)
633 {
634 ++m_nonfinite;
635 }
636 continue;
637 }
638 }
639
640 // Step 4: the order-unity vector g, with V = R g. Forming g first is what makes
641 // the representability test meaningful: an orthogonal rotation can spread one
642 // over-large local component across three representable global ones, so the test
643 // has to be applied to the components that are actually returned.
644 point_type g;
645 g[0]= sqrtKappa * theta_perp * nx;
646 g[1]= sqrtKappa * theta_perp * ny;
647 g[2]= sqrtKappa * theta_par * nz;
648 if (!axisAligned)
649 {
650 g= rotate_from_fieldAligned_frame(g, ub);
651 }
652
653 // Step 5: decide representability from log|V_j| = log R + log|g_j|, and only
654 // then exponentiate. Where the radius itself is representable the product
655 // R * g_j keeps a full working-precision result; the per-component exponential
656 // is used only where it is not.
657 //
658 // A rotation is orthogonal, so |g_j| <= |g| <= sqrt(kappa) max(theta_perp,
659 // theta_par) =: gBound whatever ub is. When log R + log gBound is below the
660 // type's limit every component is representable, and the three per-component
661 // logarithms can be skipped entirely. That is the overwhelmingly common case;
662 // the exact per-component test below runs only near the limit.
663 const bool radiusRepresentable= (logR <= logMax);
664 const RealType R= radiusRepresentable ? std::exp(logR) : RealType(0.0);
665
666 point_type point;
667 bool anyOverflow= false;
668 if (radiusRepresentable && logR + logGBound < logMax)
669 {
670 point[0]= R * g[0];
671 point[1]= R * g[1];
672 point[2]= R * g[2];
673 }
674 else
675 {
676 for (int j= 0; j < 3; ++j)
677 {
678 if (g[j] == RealType(0.0))
679 {
680 point[j]= RealType(0.0);
681 continue;
682 }
683 const RealType logComponent= logR + std::log(std::abs(g[j]));
684 if (logComponent >= logMax)
685 {
686 anyOverflow= true;
687 point[j]= (g[j] < RealType(0.0))
688 ? -std::numeric_limits<RealType>::infinity()
689 : std::numeric_limits<RealType>::infinity();
690 continue;
691 }
692 if (radiusRepresentable)
693 {
694 point[j]= R * g[j];
695 }
696 else
697 {
698 const RealType mag= std::exp(logComponent);
699 point[j]= (g[j] < RealType(0.0)) ? -mag : mag;
700 }
701 }
702 }
703
704 if (anyOverflow)
705 {
706 // Honest overflow: the intended velocity has no representation in RealType.
707 // No formation can rescue it, so it is returned as a non-finite vector and
708 // counted, never silently redrawn. In capped mode it is unreachable, having
709 // already been rejected by the box test above.
710 ++m_nonfinite;
711 }
712
713 return point;
714 }
715
716 throw std::runtime_error("failed to sample velocity under max_normalized_velocity cap: "
717 "rejection sampling exceeded internal attempt limit");
718 }

720 // ====================================================================
721 // Geometry helper
722 // ====================================================================
723 // Transform from velocity components (v_perp, v_par) to vector velocity (vx, vy, vz)
724 // given the parallel direction ub.
725 point_type rotate_from_fieldAligned_frame(const point_type &local, const point_type &ub) const
726 {
727 const RealType kZero= RealType(0.0);
728 const RealType kOne= RealType(1.0);
729
730 // Normalize ub to get e3
731 RealType ubNorm2= ub[0] * ub[0] + ub[1] * ub[1] + ub[2] * ub[2];
732 if (approximatelyEqual(ubNorm2, kZero))
733 {
734 throw std::invalid_argument("ub must be a non-zero vector");
735 }
736
737 RealType ubNorm= std::sqrt(ubNorm2);
738
739 point_type e3;
740 e3[0]= ub[0] / ubNorm;
741 e3[1]= ub[1] / ubNorm;
742 e3[2]= ub[2] / ubNorm;
743
744 // Build e2 perpendicular to e3
745 point_type e2({kOne, kOne, kOne});
746
747 int maxcomp= 0;
748 if (std::abs(e3[1]) > std::abs(e3[maxcomp]))
749 {
750 maxcomp= 1;
751 }
752 if (std::abs(e3[2]) > std::abs(e3[maxcomp]))
753 {
754 maxcomp= 2;
755 }
756
757 RealType sum= e3[0] + e3[1] + e3[2];
758 e2[maxcomp]= kOne - sum / e3[maxcomp];
759
760 RealType e2Norm2= e2[0] * e2[0] + e2[1] * e2[1] + e2[2] * e2[2];
761 RealType e2Norm= std::sqrt(e2Norm2);
762 e2[0]/= e2Norm;
763 e2[1]/= e2Norm;
764 e2[2]/= e2Norm;
765
766 // e1 = e2 x e3
767 point_type e1;
768 e1[0]= e2[1] * e3[2] - e2[2] * e3[1];
769 e1[1]= e2[2] * e3[0] - e2[0] * e3[2];
770 e1[2]= e2[0] * e3[1] - e2[1] * e3[0];
771
772 // Transform local coordinates to Cartesian coordinates
773 point_type point;
774 point[0]= local[0] * e1[0] + local[1] * e2[0] + local[2] * e3[0];
775 point[1]= local[0] * e1[1] + local[1] * e2[1] + local[2] * e3[1];
776 point[2]= local[0] * e1[2] + local[1] * e2[2] + local[2] * e3[2];
777
778 return point;
779 }
