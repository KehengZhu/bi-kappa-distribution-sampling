29// This sampler decides what it may return by materializing a value and asking whether it is
30// finite, and no_cap() is +infinity. `-ffast-math` (which defines __FAST_MATH__) licenses a
31// compiler to assume that no infinity or NaN occurs, and so to fold `mag <= max()` to `true`:
32// an overflowed component could then be returned as if it were a number, n_nonfinite() would
33// read zero, and no_cap() would not be reliable. A finite cap (the default of 20) still
34// rejects such draws, because it compares against a finite bound. The header therefore
35// compiles under fast-math but says so; define BI_KAPPA_ALLOW_FAST_MATH to silence the note.
36#if defined(__FAST_MATH__) && !defined(BI_KAPPA_ALLOW_FAST_MATH)
37#pragma message("bi_kappa_distribution.H: compiled with -ffast-math; no_cap() and the overflow " \
38 "counters are unreliable in this mode. Define BI_KAPPA_ALLOW_FAST_MATH to " \
39 "silence this note.")
40#endif

66template <typename GeneratorType> inline int engineBits()
67{
68 unsigned long long range= static_cast<unsigned long long>(GeneratorType::max()) -
69 static_cast<unsigned long long>(GeneratorType::min());
70 int bits= 0;
71 while (range >= 1ULL)
72 {
73 ++bits;
74 range>>= 1;
75 }
76 return bits;
77}

93template <typename RealType, typename GeneratorType> RealType openCanonical(GeneratorType &gen)
94{
95 const int bits= std::numeric_limits<RealType>::digits - 1;
96 const int per= engineBits<GeneratorType>();
97 unsigned long long k= 0;
98 int have= 0;
99 while (have < bits)
100 {
101 const unsigned long long word= static_cast<unsigned long long>(gen()) -
102 static_cast<unsigned long long>(GeneratorType::min());
103 const int take= (bits - have < per) ? (bits - have) : per;
104 k= (k << take) | (word & ((1ULL << take) - 1ULL));
105 have+= take;
106 }
107 return (static_cast<RealType>(k) + RealType(0.5)) * std::ldexp(RealType(1.0), -bits);
108}

113template <typename RealType, typename GeneratorType>
114RealType polarNormal(GeneratorType &gen, RealType &cache, bool &cached)
115{
116 if (cached)
117 {
118 cached= false;
119 return cache;
120 }
121 RealType x, y, s;
122 do
123 {
124 x= RealType(2.0) * openCanonical<RealType, GeneratorType>(gen) - RealType(1.0);
125 y= RealType(2.0) * openCanonical<RealType, GeneratorType>(gen) - RealType(1.0);
126 s= x * x + y * y;
127 } while (s >= RealType(1.0));
128 // s == 0 is unreachable: openCanonical never returns exactly 1/2, so neither x nor y is
129 // ever exactly zero.
130 const RealType f= std::sqrt(RealType(-2.0) * std::log(s) / s);
131 cache= y * f;
132 cached= true;
133 return x * f;
134}

139template <typename RealType, typename GeneratorType>
140RealType marsagliaTsangGamma(RealType shape, GeneratorType &gen, RealType &ncache, bool &ncached)
141{
142 const RealType d= shape - RealType(1.0) / RealType(3.0);
143 const RealType c= RealType(1.0) / std::sqrt(RealType(9.0) * d);
144 for (;;)
145 {
146 RealType x, v;
147 do
148 {
149 x= polarNormal<RealType, GeneratorType>(gen, ncache, ncached);
150 v= RealType(1.0) + c * x;
151 } while (v <= RealType(0.0));
152 v= v * v * v;
153 const RealType u= openCanonical<RealType, GeneratorType>(gen);
154 const RealType xx= x * x;
155 if (u < RealType(1.0) - RealType(0.0331) * xx * xx)
156 {
157 return d * v;
158 }
159 if (std::log(u) < RealType(0.5) * xx + d * (RealType(1.0) - v + std::log(v)))
160 {
161 return d * v;
162 }
163 }
164}

193template <typename RealType, typename GeneratorType>
194RealType logGamma(RealType a, GeneratorType &gen, RealType &ncache, bool &ncached)
195{
196 const RealType y=
197 marsagliaTsangGamma<RealType, GeneratorType>(a + RealType(1.0), gen, ncache, ncached);
198 const RealType u= openCanonical<RealType, GeneratorType>(gen);
199 return std::log(y) + std::log(u) / a;
200}

243template <typename RealType>
244bool materializeComponents(RealType logR, const std::array<RealType, 3> &g,
245 std::array<RealType, 3> &point)
246{
247 const RealType maxFinite= std::numeric_limits<RealType>::max();
248 const RealType rTry= std::exp(logR);
249 // `false` for +infinity and for NaN, both of which fail every comparison.
250 const bool radiusRepresentable= (rTry <= maxFinite);
251 const RealType R= radiusRepresentable ? rTry : RealType(0.0);
252
253 bool anyOverflow= false;
254 for (int j= 0; j < 3; ++j)
255 {
256 if (g[j] == RealType(0.0))
257 {
258 point[j]= RealType(0.0);
259 continue;
260 }
261 const RealType absG= std::abs(g[j]);
262 // What decides representability is the number the caller will receive. A
263 // magnitude that rounds past `max()` becomes an infinity here and is counted;
264 // deciding from a logarithm instead would be wrong in both directions at once.
265 const RealType mag= radiusRepresentable ? R * absG
266 : std::exp(logR + std::log(absG));
267 if (!(mag <= maxFinite))
268 {
269 anyOverflow= true;
270 point[j]= (g[j] < RealType(0.0)) ? -std::numeric_limits<RealType>::infinity()
271 : std::numeric_limits<RealType>::infinity();
272 continue;
273 }
274 point[j]= (g[j] < RealType(0.0)) ? -mag : mag;
275 }
276 return anyOverflow;
277}

346 // The cap defaults to 20 thermal speeds, as in release 1.0.0, so that code written
347 // against 1.0.0 samples the same distribution. Pass no_cap() (= +infinity) for the
348 // uncapped distribution.
349 param_type(RealType kappa= RealType(2.0), RealType theta_perp= RealType(1.0),
350 RealType theta_par= RealType(1.0),
351 point_type ub= point_type({RealType(0.0), RealType(0.0), RealType(1.0)}),
352 RealType max_normalized_velocity= RealType(20.0))
353 : m_kappa(kappa), m_theta_perp(theta_perp), m_theta_par(theta_par), m_ub(ub),
354 m_maxNormalizedVelocity(max_normalized_velocity)
355 {
356 // Each test is written so that NaN fails it. `x <= 0.5` is false for a NaN,
357 // so the natural spelling admits one, and a NaN parameter then propagates into
358 // log R, where it can no longer be told from an unrepresentable velocity: the
359 // sampler would report a non-finite draw and count it as an overflow, which it
360 // is not. Rejecting the parameter is the only point at which the two are
361 // still distinguishable.
362 if (!(m_kappa > RealType(0.5)))
363 {
364 throw std::invalid_argument("kappa must be > 0.5 for normalization");
365 }
366 if (!(m_theta_perp > RealType(0.0)) || !(m_theta_par > RealType(0.0)) ||
367 !(m_theta_perp <= std::numeric_limits<RealType>::max()) ||
368 !(m_theta_par <= std::numeric_limits<RealType>::max()))
369 {
370 throw std::invalid_argument(
371 "Temperature parameters must be positive and finite");
372 }
373 {
374 // The field direction is validated here rather than inside the rotation, so
375 // that an axis-aligned-looking NaN cannot slip past the fast path.
376 RealType n2= RealType(0.0);
377 for (int j= 0; j < 3; ++j)
378 {
379 if (!(std::abs(m_ub[j]) <= std::numeric_limits<RealType>::max()))
380 {
381 throw std::invalid_argument("ub components must be finite");
382 }
383 n2+= m_ub[j] * m_ub[j];
384 }
385 if (!(n2 > RealType(0.0)))
386 {
387 throw std::invalid_argument("ub must be a non-zero vector");
388 }
389 }
390 // A positive finite value selects the capped mode. no_cap() (= +inf)
391 // selects the untruncated mode, in which no acceptance-rejection loop
392 // runs at all and samples follow the full bi-Kappa law.
393 if (!(m_maxNormalizedVelocity > RealType(0.0)))
394 {
395 throw std::invalid_argument(
396 "max_normalized_velocity must be positive, or no_cap() for the "
397 "untruncated distribution");
398 }
399 }

659 template <typename GeneratorType>
660 result_type operator()(GeneratorType &gen, const param_type &params)
661 {
662 const RealType kHalf= RealType(0.5);
663 const RealType kOne= RealType(1.0);
664 const int kMaxCapRejectTries= 1000000;
665
666 RealType kappa= params.kappa();
667 RealType theta_perp= params.theta_perp();
668 RealType theta_par= params.theta_par();
669 point_type ub= params.ub();
670 RealType maxNormalizedVelocity= params.max_normalized_velocity();
671
672 const RealType a= kappa - kHalf;
673 const RealType sqrtKappa= std::sqrt(kappa);
674
675 // Untruncated mode runs the core mapping exactly once; capped mode redraws
676 // until the sample falls inside the box.
677 const bool capped= params.capped();
678 const int maxTries= capped ? kMaxCapRejectTries : 1;
679
680 const bool axisAligned= approximatelyEqual(ub[0], RealType(0.0)) &&
681 approximatelyEqual(ub[1], RealType(0.0)) &&
682 approximatelyEqual(ub[2], RealType(1.0));
683
684 // Four logarithms that only the capped branch reads, and so are only taken when
685 // there is a cap. Untruncated mode decides representability from the components
686 // themselves and needs no logarithm at all outside step 1; computing these anyway
687 // cost two `std::log` calls on every untruncated draw, which `-fmath-errno` (the
688 // default) forbids a compiler from removing.
689 //
690 // `logMax` is a screening quantity only: the cap branch uses it to decide whether a
691 // rejected draw is near enough to the type's limit to be worth materializing.
692 // Representability itself is never decided from a logarithm -- see
693 // bikappa_detail::materializeComponents.
694 //
695 // `logGBound` bounds the *velocity* from its radius alone:
696 // |g_j| <= |g| <= sqrt(kappa) max(theta_perp, theta_par) for every direction and
697 // every rotation, because a rotation preserves the norm. That is what the capped
698 // branch needs and what the normalized coordinate the cap test is written in cannot
699 // give it. It is finite because define() rejects a non-positive or non-finite
700 // theta, so the screen below is never comparing against a NaN.
701 const RealType logMax= capped ? std::log(std::numeric_limits<RealType>::max())
702 : RealType(0.0);
703 const RealType logCap= capped ? std::log(maxNormalizedVelocity) : RealType(0.0);
704 const RealType logSqrtKappa= capped ? std::log(sqrtKappa) : RealType(0.0);
705 const RealType logGBound=
706 capped ? logSqrtKappa + std::log(std::max(theta_perp, theta_par)) : RealType(0.0);
707
708 int tries= 0;
709 for (tries= 0; tries < maxTries; ++tries)
710 {
711 ++m_attempts;
712
713 // Step 1: the radius, carried as its logarithm.
714 //
715 // R = sqrt(X1)/sqrt(X2) with X1 ~ Gamma(3/2,1) and X2 ~ Gamma(a,1), a = kappa-1/2.
716 // As kappa -> 1/2 the shape a -> 0 and X2 underflows to exactly zero, which no
717 // rearrangement of sqrt(X1)/sqrt(X2) can survive. X2 is therefore never formed:
718 // log X2 comes from the shape-boosting identity and
719 //
720 // log R = (log X1 - log X2) / 2
721 //
722 // stays finite and accurate wherever the mathematical radius exists. X1 has
723 // shape 3/2 > 1, so it is drawn directly and its logarithm is well conditioned.
724 const RealType x1=
725 bikappa_detail::marsagliaTsangGamma<RealType, GeneratorType>(
726 RealType(1.5), gen, m_normalCache, m_normalCached);
727 const RealType logX2=
728 bikappa_detail::logGamma<RealType, GeneratorType>(a, gen, m_normalCache,
729 m_normalCached);
730 const RealType logR= kHalf * (std::log(x1) - logX2);
731
732 // Step 2: a direction uniform on S^2, by the rejection method of Marsaglia
733 // (1972), Ann. Math. Statist. 43(2), 645-646. Draw a point uniformly in the
734 // unit disc and lift it; the lift is area-preserving, so the result is exactly
735 // uniform on the sphere.
736 //
737 // The textbook alternative -- cos(theta) uniform on (-1,1) and phi uniform on
738 // (0, 2pi) -- is equally correct as mathematics, and is what this loader used
739 // before 2.0.0. It needs sin(phi) and cos(phi) of the same argument, and a
740 // compiler is free to fuse that pair into one combined routine. Clang does, on
741 // Darwin, into __sincos_stret, whose sine differs from the standalone sin by one
742 // unit in the last place for roughly one argument in a thousand; GCC calls the
743 // two separately. Two builds of this header then return different numbers for
744 // the same seed, and the difference belongs to the compiler rather than to the
745 // sampler. The rejection method uses only sqrt, which IEEE-754 requires to be
746 // correctly rounded, so the direction is identical under every compiler. It is
747 // also faster here, despite averaging 4/pi = 1.273 attempts per direction.
748 RealType d1, d2, dsq;
749 do
750 {
751 d1= RealType(2.0) *
752 bikappa_detail::openCanonical<RealType, GeneratorType>(gen) - kOne;
753 d2= RealType(2.0) *
754 bikappa_detail::openCanonical<RealType, GeneratorType>(gen) - kOne;
755 dsq= d1 * d1 + d2 * d2;
756 } while (dsq >= kOne);
757 // dsq == 0 is unreachable: openCanonical never returns exactly 1/2, so neither
758 // d1 nor d2 is ever exactly zero.
759 const RealType dscale= RealType(2.0) * std::sqrt(kOne - dsq);
760
761 const RealType nx= d1 * dscale;
762 const RealType ny= d2 * dscale;
763 const RealType nz= kOne - RealType(2.0) * dsq;
764
765 // Step 3: the cap, tested in the field-aligned frame before anything is
766 // exponentiated. This is the same predicate the linear form applies --
767 // |point_j| / theta_j = sqrt(kappa) R |n_j| <= lambda -- moved into the log
768 // domain so that an intended sample far outside the box is rejected rather
769 // than overflowed first and rejected afterwards. Only the largest component
770 // has to be examined, so this costs one logarithm rather than three.
771 if (capped)
772 {
773 const RealType maxAbsN=
774 std::max(std::abs(nx), std::max(std::abs(ny), std::abs(nz)));
775 const RealType logLargest= logR + logSqrtKappa + std::log(maxAbsN);
776 if (logLargest > logCap)
777 {
778 // An intended sample that is outside the box and also unrepresentable is
779 // still counted, so a caller can tell the two apart. Two things have to
780 // be right here.
781 //
782 // *What* is tested. Representability is a property of the velocity
783 // `R g_j`, not of the normalized coordinate `sqrt(kappa) R |n_j|` the cap
784 // test just used: the two differ by `theta` and by the rotation. The
785 // ratio is bounded by `sqrt(3) max(theta_perp, theta_par)/min(...)`,
786 // since `|w|/sqrt(3) <= max_j |w_j| <= |w|` -- at
787 // `theta_par/theta_perp = 2` that is 3.46. The rotation concentrates the
788 // largest component as readily as it spreads it, which is why the bound
789 // is not simply the theta ratio. The counter must answer the question
790 // n_nonfinite() documents, which is about the velocity.
791 //
792 // *How* it is tested. By materializing, for the same reason as step 5.
793 // The exponential is worth avoiding here -- in capped mode this is the
794 // hot branch -- so `logGBound` screens first. It is an upper bound on
795 // log|V_j| that holds for every direction and every rotation, so a draw a
796 // whole natural-log unit below the limit by that bound cannot overflow.
797 //
798 // The screen is sound, not merely generous. Its error is dominated by
799 // half an ulp of `logR + logGBound` at the limit, which is 88.72 * 2^-24
800 // = 5.3e-6 in `float` and 709.78 * 2^-53 = 7.9e-14 in `double`; with the
801 // libm and normalization terms the worst case is about 1.1e-5 and
802 // 1.0e-13. One natural-log unit exceeds those by 9.4e4 and 1.0e13.
803 //
804 // How often it fires is a property of the configuration, not a constant:
805 // about 7e-7 of attempts at double kappa = 0.51, but 17 % at float
806 // kappa = 0.51 and 87 % at double kappa = 0.5001, where most of the law
807 // genuinely lies outside the type. Where it fires often, it is doing
808 // the work it exists for.
809 if (logR + logGBound > logMax - kOne)
810 {
811 point_type gRej= fieldAlignedG(sqrtKappa, theta_perp, theta_par,
812 nx, ny, nz);
813 if (!axisAligned)
814 {
815 gRej= rotateFromFieldAlignedFrame(gRej, ub);
816 }
817 point_type scratch;
818 if (bikappa_detail::materializeComponents<RealType>(logR, gRej,
819 scratch))
820 {
821 ++m_nonfinite;
822 }
823 }
824 continue;
825 }
826 }
827
828 // Step 4: the order-unity vector g, with V = R g. Forming g first is what makes
829 // the representability test meaningful: an orthogonal rotation can spread one
830 // over-large local component across three representable global ones, so the test
831 // has to be applied to the components that are actually returned.
832 point_type g= fieldAlignedG(sqrtKappa, theta_perp, theta_par, nx, ny, nz);
833 if (!axisAligned)
834 {
835 g= rotateFromFieldAlignedFrame(g, ub);
836 }
837
838 // Step 5: materialize the components and decide representability from the
839 // components themselves, never from their logarithms. The whole of that
840 // decision lives in bikappa_detail::materializeComponents, which documents why
841 // a logarithmic test is wrong in both directions; the capped branch above calls
842 // the same routine, so the counter and the return can never disagree.
843 point_type point;
844 const bool anyOverflow=
845 bikappa_detail::materializeComponents<RealType>(logR, g, point);
846
847 if (anyOverflow)
848 {
849 // True overflow: the intended velocity has no representation in RealType.
850 // No formation can rescue it, so it is returned as a non-finite vector and
851 // counted, never silently redrawn. In capped mode this is unreachable for
852 // every cap a caller would write: an accepted draw has
853 // sqrt(kappa) R max_j|n_j| <= lambda, hence
854 // max_j |g_j| R <= sqrt(3) lambda max(theta_perp, theta_par), so reaching it
855 // would take lambda max(theta) above 1.97e38 in `float` or 1.04e308 in
856 // `double` -- a box larger than the type.
857 ++m_nonfinite;
858 }
859
860 return point;
861 }

876
880 static point_type rotateFromFieldAlignedFrame(const point_type &local, const point_type &ub)
881 {
882 const RealType kZero= RealType(0.0);
883 const RealType kOne= RealType(1.0);
884
885 // Normalize ub to get e3
886 RealType ubNorm2= ub[0] * ub[0] + ub[1] * ub[1] + ub[2] * ub[2];
887 if (approximatelyEqual(ubNorm2, kZero))
888 {
889 throw std::invalid_argument("ub must be a non-zero vector");
890 }
891
892 RealType ubNorm= std::sqrt(ubNorm2);
893
894 point_type e3;
895 e3[0]= ub[0] / ubNorm;
896 e3[1]= ub[1] / ubNorm;
897 e3[2]= ub[2] / ubNorm;
898
899 // Build e2 perpendicular to e3
900 point_type e2;
901 e2[0]= kOne;
902 e2[1]= kOne;
903 e2[2]= kOne;
904
905 int maxcomp= 0;
906 if (std::abs(e3[1]) > std::abs(e3[maxcomp]))
907 {
908 maxcomp= 1;
909 }
910 if (std::abs(e3[2]) > std::abs(e3[maxcomp]))
911 {
912 maxcomp= 2;
913 }
914
915 RealType sum= e3[0] + e3[1] + e3[2];
916 e2[maxcomp]= kOne - sum / e3[maxcomp];
917
918 RealType e2Norm2= e2[0] * e2[0] + e2[1] * e2[1] + e2[2] * e2[2];
919 RealType e2Norm= std::sqrt(e2Norm2);
920 e2[0]/= e2Norm;
921 e2[1]/= e2Norm;
922 e2[2]/= e2Norm;
923
924 // e1 = e2 x e3
925 point_type e1;
926 e1[0]= e2[1] * e3[2] - e2[2] * e3[1];
927 e1[1]= e2[2] * e3[0] - e2[0] * e3[2];
928 e1[2]= e2[0] * e3[1] - e2[1] * e3[0];
929
930 // Transform local coordinates to Cartesian coordinates
931 point_type point;
932 point[0]= local[0] * e1[0] + local[1] * e2[0] + local[2] * e3[0];
933 point[1]= local[0] * e1[1] + local[1] * e2[1] + local[2] * e3[1];
934 point[2]= local[0] * e1[2] + local[1] * e2[2] + local[2] * e3[2];
935
936 return point;
937 }
