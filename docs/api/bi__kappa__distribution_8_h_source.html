28// This sampler decides what it may return by materializing a value and asking whether it is
29// finite, and no_cap() is +infinity. `-ffast-math` (which defines __FAST_MATH__) licenses a
30// compiler to assume that no infinity or NaN occurs, and so to fold `mag <= max()` to `true`:
31// an overflowed component could then be returned as if it were a number, n_nonfinite() would
32// read zero, and no_cap() would not be reliable. A finite cap (the default of 20) still
33// rejects such draws, because it compares against a finite bound. The header therefore
34// compiles under fast-math but says so; define BI_KAPPA_ALLOW_FAST_MATH to silence the note.
35#if defined(__FAST_MATH__) && !defined(BI_KAPPA_ALLOW_FAST_MATH)
36#pragma message("bi_kappa_distribution.H: compiled with -ffast-math; no_cap() and the overflow " \
37 "counters are unreliable in this mode. Define BI_KAPPA_ALLOW_FAST_MATH to " \
38 "silence this note.")
39#endif

106template <typename GeneratorType> inline int engineBits()
107{
108 unsigned long long range= static_cast<unsigned long long>(GeneratorType::max()) -
109 static_cast<unsigned long long>(GeneratorType::min());
110 int bits= 0;
111 while (range >= 1ULL)
112 {
113 ++bits;
114 range>>= 1;
115 }
116 return bits;
117}

133template <typename RealType, typename GeneratorType> RealType openCanonical(GeneratorType &gen)
134{
135 const int bits= std::numeric_limits<RealType>::digits - 1;
136 const int per= engineBits<GeneratorType>();
137 unsigned long long k= 0;
138 int have= 0;
139 while (have < bits)
140 {
141 const unsigned long long word= static_cast<unsigned long long>(gen()) -
142 static_cast<unsigned long long>(GeneratorType::min());
143 const int take= (bits - have < per) ? (bits - have) : per;
144 k= (k << take) | (word & ((1ULL << take) - 1ULL));
145 have+= take;
146 }
147 return (static_cast<RealType>(k) + RealType(0.5)) * std::ldexp(RealType(1.0), -bits);
148}

153template <typename RealType, typename GeneratorType>
154RealType polarNormal(GeneratorType &gen, RealType &cache, bool &cached)
155{
156 if (cached)
157 {
158 cached= false;
159 return cache;
160 }
161 RealType x, y, s;
162 do
163 {
164 x= RealType(2.0) * openCanonical<RealType, GeneratorType>(gen) - RealType(1.0);
165 y= RealType(2.0) * openCanonical<RealType, GeneratorType>(gen) - RealType(1.0);
166 s= x * x + y * y;
167 } while (s >= RealType(1.0));
168 // s == 0 is unreachable: openCanonical never returns exactly 1/2, so neither x nor y is
169 // ever exactly zero.
170 const RealType f= std::sqrt(RealType(-2.0) * std::log(s) / s);
171 cache= y * f;
172 cached= true;
173 return x * f;
174}

179template <typename RealType, typename GeneratorType>
180RealType marsagliaTsangGamma(RealType shape, GeneratorType &gen, RealType &ncache, bool &ncached)
181{
182 const RealType d= shape - RealType(1.0) / RealType(3.0);
183 const RealType c= RealType(1.0) / std::sqrt(RealType(9.0) * d);
184 for (;;)
185 {
186 RealType x, v;
187 do
188 {
189 x= polarNormal<RealType, GeneratorType>(gen, ncache, ncached);
190 v= RealType(1.0) + c * x;
191 } while (v <= RealType(0.0));
192 v= v * v * v;
193 const RealType u= openCanonical<RealType, GeneratorType>(gen);
194 const RealType xx= x * x;
195 if (u < RealType(1.0) - RealType(0.0331) * xx * xx)
196 {
197 return d * v;
198 }
199 if (std::log(u) < RealType(0.5) * xx + d * (RealType(1.0) - v + std::log(v)))
200 {
201 return d * v;
202 }
203 }
204}

229template <typename RealType, typename GeneratorType>
230typename log_accumulator<RealType>::type logGamma(RealType a, GeneratorType &gen,
231 RealType &ncache, bool &ncached)
232{
233 typedef typename log_accumulator<RealType>::type AccumType;
234 const RealType y=
235 marsagliaTsangGamma<RealType, GeneratorType>(a + RealType(1.0), gen, ncache, ncached);
236 const RealType u= openCanonical<RealType, GeneratorType>(gen);
237 // Y, U and the shape are the working-precision values the sampler actually drew and the
238 // caller actually asked for; only the logarithms and the division are widened. Widening
239 // the shape instead of the variates would change which distribution is sampled.
240 return std::log(static_cast<AccumType>(y)) +
241 std::log(static_cast<AccumType>(u)) / static_cast<AccumType>(a);
242}

285template <typename RealType>
286bool materializeComponents(typename log_accumulator<RealType>::type logR,
287 const std::array<typename log_accumulator<RealType>::type, 3> &g,
288 std::array<RealType, 3> &point)
289{
290 typedef typename log_accumulator<RealType>::type AccumType;
291 const AccumType accumMax= std::numeric_limits<AccumType>::max();
292 const RealType maxFinite= std::numeric_limits<RealType>::max();
293 const AccumType rTry= std::exp(logR);
294 // `false` for +infinity and for NaN, both of which fail every comparison. The test is
295 // in the accumulator, because that is where the product below is formed; a `float` run
296 // reaches this branch only for a radius beyond `exp(709.78)`, far above anything a
297 // `float` component could hold.
298 const bool radiusRepresentable= (rTry <= accumMax);
299 const AccumType R= radiusRepresentable ? rTry : AccumType(0.0);
300
301 bool anyOverflow= false;
302 for (int j= 0; j < 3; ++j)
303 {
304 if (g[j] == AccumType(0.0))
305 {
306 point[j]= RealType(0.0);
307 continue;
308 }
309 const AccumType absG= std::abs(g[j]);
310 const AccumType wide= radiusRepresentable ? R * absG
311 : std::exp(logR + std::log(absG));
312 // The single rounding. What decides representability is the number the caller will
313 // receive, which is `wide` rounded to `RealType` -- so a magnitude that lies within
314 // half an ulp of `max()` rounds to `max()` and is returned, and one that rounds past
315 // it becomes an infinity here and is counted. Deciding from `wide` instead would
316 // discard a draw the type can hold; deciding from a logarithm would do it in both
317 // directions at once.
318 const RealType mag= static_cast<RealType>(wide);
319 if (!(mag <= maxFinite))
320 {
321 anyOverflow= true;
322 point[j]= (g[j] < AccumType(0.0)) ? -std::numeric_limits<RealType>::infinity()
323 : std::numeric_limits<RealType>::infinity();
324 continue;
325 }
326 point[j]= (g[j] < AccumType(0.0)) ? -mag : mag;
327 }
328 return anyOverflow;
329}

404 // The cap defaults to 20 thermal speeds, as in release 1.0.0, so that code written
405 // against 1.0.0 samples the same distribution. Pass no_cap() (= +infinity) for the
406 // uncapped distribution.
407 param_type(RealType kappa= RealType(2.0), RealType theta_perp= RealType(1.0),
408 RealType theta_par= RealType(1.0),
409 point_type ub= point_type({RealType(0.0), RealType(0.0), RealType(1.0)}),
410 RealType max_normalized_velocity= RealType(20.0))
411 : m_kappa(kappa), m_theta_perp(theta_perp), m_theta_par(theta_par), m_ub(ub),
412 m_maxNormalizedVelocity(max_normalized_velocity)
413 {
414 // Each test is written so that NaN fails it. `x <= 0.5` is false for a NaN,
415 // so the natural spelling admits one, and a NaN parameter then propagates into
416 // log R, where it can no longer be told from an unrepresentable velocity: the
417 // sampler would report a non-finite draw and count it as an overflow, which it
418 // is not. Rejecting the parameter is the only point at which the two are
419 // still distinguishable.
420 if (!(m_kappa > RealType(0.5)))
421 {
422 throw std::invalid_argument("kappa must be > 0.5 for normalization");
423 }
424 if (!(m_theta_perp > RealType(0.0)) || !(m_theta_par > RealType(0.0)) ||
425 !(m_theta_perp <= std::numeric_limits<RealType>::max()) ||
426 !(m_theta_par <= std::numeric_limits<RealType>::max()))
427 {
428 throw std::invalid_argument(
429 "Temperature parameters must be positive and finite");
430 }
431 {
432 // The field direction is validated here rather than inside the rotation, so
433 // that an axis-aligned-looking NaN cannot slip past the fast path.
434 RealType n2= RealType(0.0);
435 for (int j= 0; j < 3; ++j)
436 {
437 if (!(std::abs(m_ub[j]) <= std::numeric_limits<RealType>::max()))
438 {
439 throw std::invalid_argument("ub components must be finite");
440 }
441 n2+= m_ub[j] * m_ub[j];
442 }
443 if (!(n2 > RealType(0.0)))
444 {
445 throw std::invalid_argument("ub must be a non-zero vector");
446 }
447 }
448 // A positive finite value selects the capped mode. no_cap() (= +inf)
449 // selects the untruncated mode, in which no acceptance-rejection loop
450 // runs at all and samples follow the full bi-Kappa law.
451 if (!(m_maxNormalizedVelocity > RealType(0.0)))
452 {
453 throw std::invalid_argument(
454 "max_normalized_velocity must be positive, or no_cap() for the "
455 "untruncated distribution");
456 }
457 }

717 template <typename GeneratorType>
718 result_type operator()(GeneratorType &gen, const param_type &params)
719 {
720 typedef typename bikappa_detail::log_accumulator<RealType>::type AccumType;
721 const RealType kHalf= RealType(0.5);
722 const RealType kOne= RealType(1.0);
723 const AccumType kHalfA= AccumType(0.5);
724 const AccumType kOneA= AccumType(1.0);
725 const int kMaxCapRejectTries= 1000000;
726
727 RealType kappa= params.kappa();
728 RealType theta_perp= params.theta_perp();
729 RealType theta_par= params.theta_par();
730 point_type ub= params.ub();
731 RealType maxNormalizedVelocity= params.max_normalized_velocity();
732
733 const RealType a= kappa - kHalf;
734 // sqrt(kappa) is not a variate; it is part of the deterministic map from the drawn
735 // variates to the returned velocity, so it is formed in the accumulator with
736 // everything else on that path.
737 const AccumType sqrtKappa= std::sqrt(static_cast<AccumType>(kappa));
738
739 // Untruncated mode runs the core mapping exactly once; capped mode redraws
740 // until the sample falls inside the box.
741 const bool capped= params.capped();
742 const int maxTries= capped ? kMaxCapRejectTries : 1;
743
744 const bool axisAligned= approximatelyEqual(ub[0], RealType(0.0)) &&
745 approximatelyEqual(ub[1], RealType(0.0)) &&
746 approximatelyEqual(ub[2], RealType(1.0));
747
748 // Four logarithms that only the capped branch reads, and so are only taken when
749 // there is a cap. Untruncated mode decides representability from the components
750 // themselves and needs no logarithm at all outside step 1; computing these anyway
751 // cost two `std::log` calls on every untruncated draw, which `-fmath-errno` (the
752 // default) forbids a compiler from removing.
753 //
754 // `logMax` is a screening quantity only: the cap branch uses it to decide whether a
755 // rejected draw is near enough to the type's limit to be worth materializing.
756 // Representability itself is never decided from a logarithm -- see
757 // bikappa_detail::materializeComponents.
758 //
759 // `logGBound` bounds the *velocity* from its radius alone:
760 // |g_j| <= |g| <= sqrt(kappa) max(theta_perp, theta_par) for every direction and
761 // every rotation, because a rotation preserves the norm. That is what the capped
762 // branch needs and what the normalized coordinate the cap test is written in cannot
763 // give it. It is finite because define() rejects a non-positive or non-finite
764 // theta, so the screen below is never comparing against a NaN.
765 // All four are formed in the accumulator, so that the screen and the cap predicate
766 // resolve the same margins the radius does.
767 const AccumType logMax=
768 capped ? std::log(static_cast<AccumType>(std::numeric_limits<RealType>::max()))
769 : AccumType(0.0);
770 const AccumType logCap=
771 capped ? std::log(static_cast<AccumType>(maxNormalizedVelocity)) : AccumType(0.0);
772 const AccumType logSqrtKappa= capped ? std::log(sqrtKappa) : AccumType(0.0);
773 const AccumType logGBound=
774 capped ? logSqrtKappa +
775 std::log(static_cast<AccumType>(std::max(theta_perp, theta_par)))
776 : AccumType(0.0);
777
778 int tries= 0;
779 for (tries= 0; tries < maxTries; ++tries)
780 {
781 ++m_attempts;
782
783 // Step 1: the radius, carried as its logarithm.
784 //
785 // R = sqrt(X1)/sqrt(X2) with X1 ~ Gamma(3/2,1) and X2 ~ Gamma(a,1), a = kappa-1/2.
786 // As kappa -> 1/2 the shape a -> 0 and X2 underflows to exactly zero, which no
787 // rearrangement of sqrt(X1)/sqrt(X2) can survive. X2 is therefore never formed:
788 // log X2 comes from the shape-boosting identity and
789 //
790 // log R = (log X1 - log X2) / 2
791 //
792 // stays finite and accurate wherever the mathematical radius exists. X1 has
793 // shape 3/2 > 1, so it is drawn directly and its logarithm is well conditioned.
794 const RealType x1=
795 bikappa_detail::marsagliaTsangGamma<RealType, GeneratorType>(
796 RealType(1.5), gen, m_normalCache, m_normalCached);
797 const AccumType logX2=
798 bikappa_detail::logGamma<RealType, GeneratorType>(a, gen, m_normalCache,
799 m_normalCached);
800 const AccumType logR=
801 kHalfA * (std::log(static_cast<AccumType>(x1)) - logX2);
802
803 // Step 2: a direction uniform on S^2, by the rejection method of Marsaglia
804 // (1972), Ann. Math. Statist. 43(2), 645-646. Draw a point uniformly in the
805 // unit disc and lift it; the lift is area-preserving, so the result is exactly
806 // uniform on the sphere.
807 //
808 // The textbook alternative -- cos(theta) uniform on (-1,1) and phi uniform on
809 // (0, 2pi) -- is equally correct as mathematics, and is what this loader used
810 // before 2.0.0. It needs sin(phi) and cos(phi) of the same argument, and a
811 // compiler is free to fuse that pair into one combined routine. Clang does, on
812 // Darwin, into __sincos_stret, whose sine differs from the standalone sin by one
813 // unit in the last place for roughly one argument in a thousand; GCC calls the
814 // two separately. Two builds of this header then return different numbers for
815 // the same seed, and the difference belongs to the compiler rather than to the
816 // sampler. The rejection method uses only sqrt, which IEEE-754 requires to be
817 // correctly rounded, so the direction is identical under every compiler. It is
818 // also faster here, despite averaging 4/pi = 1.273 attempts per direction.
819 //
820 // The two coordinates are the variates and are drawn in RealType; the disc test
821 // and the lift are the deterministic part and are evaluated in the accumulator.
822 // For a `float` run that also makes the test exact -- `d1*d1 + d2*d2` of two
823 // `float`s needs 49 bits, which `double` holds and `float` does not, so the
824 // `float` form could accept a point outside the disc or reject one inside it.
825 RealType d1, d2;
826 AccumType dsq;
827 do
828 {
829 d1= RealType(2.0) *
830 bikappa_detail::openCanonical<RealType, GeneratorType>(gen) - kOne;
831 d2= RealType(2.0) *
832 bikappa_detail::openCanonical<RealType, GeneratorType>(gen) - kOne;
833 dsq= static_cast<AccumType>(d1) * static_cast<AccumType>(d1) +
834 static_cast<AccumType>(d2) * static_cast<AccumType>(d2);
835 } while (dsq >= kOneA);
836 // dsq == 0 is unreachable: openCanonical never returns exactly 1/2, so neither
837 // d1 nor d2 is ever exactly zero.
838 const AccumType dscale= AccumType(2.0) * std::sqrt(kOneA - dsq);
839
840 const AccumType nx= static_cast<AccumType>(d1) * dscale;
841 const AccumType ny= static_cast<AccumType>(d2) * dscale;
842 const AccumType nz= kOneA - AccumType(2.0) * dsq;
843
844 // Step 3: the cap, tested in the field-aligned frame before anything is
845 // exponentiated. This is the same predicate the linear form applies --
846 // |point_j| / theta_j = sqrt(kappa) R |n_j| <= lambda -- moved into the log
847 // domain so that an intended sample far outside the box is rejected rather
848 // than overflowed first and rejected afterwards. Only the largest component
849 // has to be examined, so this costs one logarithm rather than three.
850 if (capped)
851 {
852 const AccumType maxAbsN=
853 std::max(std::abs(nx), std::max(std::abs(ny), std::abs(nz)));
854 const AccumType logLargest= logR + logSqrtKappa + std::log(maxAbsN);
855 if (logLargest > logCap)
856 {
857 // An intended sample that is outside the box and also unrepresentable is
858 // still counted, so a caller can tell the two apart. Two things have to
859 // be right here.
860 //
861 // *What* is tested. Representability is a property of the velocity
862 // `R g_j`, not of the normalized coordinate `sqrt(kappa) R |n_j|` the cap
863 // test just used: the two differ by `theta` and by the rotation. The
864 // ratio is bounded by `sqrt(3) max(theta_perp, theta_par)/min(...)`,
865 // since `|w|/sqrt(3) <= max_j |w_j| <= |w|` -- at
866 // `theta_par/theta_perp = 2` that is 3.46. The rotation concentrates the
867 // largest component as readily as it spreads it, which is why the bound
868 // is not simply the theta ratio. The counter must answer the question
869 // n_nonfinite() documents, which is about the velocity.
870 //
871 // *How* it is tested. By materializing, for the same reason as step 5.
872 // The exponential is worth avoiding here -- in capped mode this is the
873 // hot branch -- so `logGBound` screens first. It is an upper bound on
874 // log|V_j| that holds for every direction and every rotation, so a draw a
875 // whole natural-log unit below the limit by that bound cannot overflow.
876 //
877 // The screen is sound, not merely generous. Its error is dominated by
878 // half an ulp of `logR + logGBound` at the limit, which is 88.72 * 2^-24
879 // = 5.3e-6 in `float` and 709.78 * 2^-53 = 7.9e-14 in `double`; with the
880 // libm and normalization terms the worst case is about 1.1e-5 and
881 // 1.0e-13. One natural-log unit exceeds those by 9.4e4 and 1.0e13.
882 //
883 // How often it fires is a property of the configuration, not a constant:
884 // about 7e-7 of attempts at double kappa = 0.51, but 17 % at float
885 // kappa = 0.51 and 87 % at double kappa = 0.5001, where most of the law
886 // genuinely lies outside the type. Where it fires often, it is doing
887 // the work it exists for.
888 if (logR + logGBound > logMax - kOneA)
889 {
890 accum_point gRej= fieldAlignedG(sqrtKappa, theta_perp, theta_par,
891 nx, ny, nz);
892 if (!axisAligned)
893 {
894 gRej= rotateFromFieldAlignedFrameIn<AccumType>(
895 gRej, widen<AccumType>(ub));
896 }
897 point_type scratch;
898 if (bikappa_detail::materializeComponents<RealType>(logR, gRej,
899 scratch))
900 {
901 ++m_nonfinite;
902 }
903 }
904 continue;
905 }
906 }
907
908 // Step 4: the order-unity vector g, with V = R g. Forming g first is what makes
909 // the representability test meaningful: an orthogonal rotation can spread one
910 // over-large local component across three representable global ones, so the test
911 // has to be applied to the components that are actually returned.
912 accum_point g= fieldAlignedG(sqrtKappa, theta_perp, theta_par, nx, ny, nz);
913 if (!axisAligned)
914 {
915 g= rotateFromFieldAlignedFrameIn<AccumType>(g, widen<AccumType>(ub));
916 }
917
918 // Step 5: materialize the components and decide representability from the
919 // components themselves, never from their logarithms. The whole of that
920 // decision lives in bikappa_detail::materializeComponents, which documents why
921 // a logarithmic test is wrong in both directions; the capped branch above calls
922 // the same routine, so the counter and the return can never disagree.
923 point_type point;
924 const bool anyOverflow=
925 bikappa_detail::materializeComponents<RealType>(logR, g, point);
926
927 if (anyOverflow)
928 {
929 // True overflow: the intended velocity has no representation in RealType.
930 // No formation can rescue it, so it is returned as a non-finite vector and
931 // counted, never silently redrawn. In capped mode this is unreachable for
932 // every cap a caller would write: an accepted draw has
933 // sqrt(kappa) R max_j|n_j| <= lambda, hence
934 // max_j |g_j| R <= sqrt(3) lambda max(theta_perp, theta_par), so reaching it
935 // would take lambda max(theta) above 1.97e38 in `float` or 1.04e308 in
936 // `double` -- a box larger than the type.
937 ++m_nonfinite;
938 }
939
940 return point;
941 }
942
943 throw std::runtime_error("failed to sample velocity under max_normalized_velocity cap: "
944 "rejection sampling exceeded internal attempt limit");
945 }

956
964 template <typename ValueType>
965 static std::array<ValueType, 3> rotateFromFieldAlignedFrameIn(
966 const std::array<ValueType, 3> &local, const std::array<ValueType, 3> &ub)
967 {
968 typedef std::array<ValueType, 3> vec_type;
969 const ValueType kZero= ValueType(0.0);
970 const ValueType kOne= ValueType(1.0);
971
972 // Normalize ub to get e3
973 ValueType ubNorm2= ub[0] * ub[0] + ub[1] * ub[1] + ub[2] * ub[2];
974 if (approximatelyEqualIn<ValueType>(ubNorm2, kZero))
975 {
976 throw std::invalid_argument("ub must be a non-zero vector");
977 }
978
979 ValueType ubNorm= std::sqrt(ubNorm2);
980
981 vec_type e3;
982 e3[0]= ub[0] / ubNorm;
983 e3[1]= ub[1] / ubNorm;
984 e3[2]= ub[2] / ubNorm;
985
986 // Build e2 perpendicular to e3
987 vec_type e2;
988 e2[0]= kOne;
989 e2[1]= kOne;
990 e2[2]= kOne;
991
992 int maxcomp= 0;
993 if (std::abs(e3[1]) > std::abs(e3[maxcomp]))
994 {
995 maxcomp= 1;
996 }
997 if (std::abs(e3[2]) > std::abs(e3[maxcomp]))
998 {
999 maxcomp= 2;
1000 }
1001
1002 ValueType sum= e3[0] + e3[1] + e3[2];
1003 e2[maxcomp]= kOne - sum / e3[maxcomp];
1004
1005 ValueType e2Norm2= e2[0] * e2[0] + e2[1] * e2[1] + e2[2] * e2[2];
1006 ValueType e2Norm= std::sqrt(e2Norm2);
1007 e2[0]/= e2Norm;
1008 e2[1]/= e2Norm;
1009 e2[2]/= e2Norm;
1010
1011 // e1 = e2 x e3
1012 vec_type e1;
1013 e1[0]= e2[1] * e3[2] - e2[2] * e3[1];
1014 e1[1]= e2[2] * e3[0] - e2[0] * e3[2];
1015 e1[2]= e2[0] * e3[1] - e2[1] * e3[0];
1016
1017 // Transform local coordinates to Cartesian coordinates
1018 vec_type point;
1019 point[0]= local[0] * e1[0] + local[1] * e2[0] + local[2] * e3[0];
1020 point[1]= local[0] * e1[1] + local[1] * e2[1] + local[2] * e3[1];
1021 point[2]= local[0] * e1[2] + local[1] * e2[2] + local[2] * e3[2];
1022
1023 return point;
1024 }
