264 template <typename GeneratorType>
265 result_type operator()(GeneratorType &gen, const param_type &params)
266 {
267 const RealType kHalf= RealType(0.5);
268 const RealType kOne= RealType(1.0);
269 const int kMaxCapRejectTries= 1000000;
270
271 RealType kappa= params.kappa();
272 RealType theta_perp= params.theta_perp();
273 RealType theta_par= params.theta_par();
274 point_type ub= params.ub();
275 RealType maxNormalizedVelocity= params.max_normalized_velocity();
276
277 typename std::gamma_distribution<RealType>::param_type distX2Param(kappa - kHalf, kOne);
278 RealType sqrtKappa= std::sqrt(kappa);
279
280 // Untruncated mode runs the core mapping exactly once; capped mode redraws
281 // until the sample falls inside the box.
282 const bool capped= params.capped();
283 const int maxTries= capped ? kMaxCapRejectTries : 1;
284
285 int tries= 0;
286 for (tries= 0; tries < maxTries; ++tries)
287 {
288 // Step 1: Sample radii.
289 // Take the two square roots separately rather than forming the ratio
290 // first: T = x1/x2 overflows for x2 near the underflow limit even where
291 // R = sqrt(T) is perfectly representable, which discards valid draws at
292 // small kappa (shape(x2) = kappa - 1/2 << 1).
293 RealType x1= m_distX1(gen);
294 RealType x2= m_distX2(gen, distX2Param);
295 RealType r= std::sqrt(x1) / std::sqrt(x2);
296
297 // Step 2: Sample directions uniformly on S^2
298 RealType cosTheta= m_distCosTheta(gen);
299 RealType sinTheta= std::sqrt(std::max(RealType(0.0), kOne - cosTheta * cosTheta));
300 RealType phi= m_distPhi(gen);
301
302 // Step 3: Compute unit vectors (u)
303 RealType ux= r * sinTheta * std::cos(phi);
304 RealType uy= r * sinTheta * std::sin(phi);
305 RealType uz= r * cosTheta;
306
307 // Step 4: Map back to v-space and apply velocity cap in local frame
308 point_type point;
309 point[0]= sqrtKappa * theta_perp * ux;
310 point[1]= sqrtKappa * theta_perp * uy;
311 point[2]= sqrtKappa * theta_par * uz;
312
313 if (capped &&
314 !withinNormalizedVelocityCap(point, theta_perp, theta_par, maxNormalizedVelocity))
315 {
316 continue;
317 }
318
319 if (approximatelyEqual(ub[0], RealType(0.0)) &&
320 approximatelyEqual(ub[1], RealType(0.0)) &&
321 approximatelyEqual(ub[2], RealType(1.0)))
322 {
323 return point;
324 }
325
326 // Step 5: rotate from local field-aligned frame to global frame
327 return rotate_from_fieldAligned_frame(point, ub);
328 }
329
330 throw std::runtime_error("failed to sample velocity under max_normalized_velocity cap: "
331 "rejection sampling exceeded internal attempt limit");
332 }
333
334 // ====================================================================
335 // Geometry helper
336 // ====================================================================
337 // Transform from velocity components (v_perp, v_par) to vector velocity (vx, vy, vz)
338 // given the parallel direction ub.
339 point_type rotate_from_fieldAligned_frame(const point_type &local, const point_type &ub) const
340 {
341 const RealType kZero= RealType(0.0);
342 const RealType kOne= RealType(1.0);
343
344 // Normalize ub to get e3
345 RealType ubNorm2= ub[0] * ub[0] + ub[1] * ub[1] + ub[2] * ub[2];
346 if (approximatelyEqual(ubNorm2, kZero))
347 {
348 throw std::invalid_argument("ub must be a non-zero vector");
349 }
350
351 RealType ubNorm= std::sqrt(ubNorm2);
352
353 point_type e3;
354 e3[0]= ub[0] / ubNorm;
355 e3[1]= ub[1] / ubNorm;
356 e3[2]= ub[2] / ubNorm;
357
358 // Build e2 perpendicular to e3
359 point_type e2({kOne, kOne, kOne});
360
361 int maxcomp= 0;
362 if (std::abs(e3[1]) > std::abs(e3[maxcomp]))
363 {
364 maxcomp= 1;
365 }
366 if (std::abs(e3[2]) > std::abs(e3[maxcomp]))
367 {
368 maxcomp= 2;
369 }
370
371 RealType sum= e3[0] + e3[1] + e3[2];
372 e2[maxcomp]= kOne - sum / e3[maxcomp];
373
374 RealType e2Norm2= e2[0] * e2[0] + e2[1] * e2[1] + e2[2] * e2[2];
375 RealType e2Norm= std::sqrt(e2Norm2);
376 e2[0]/= e2Norm;
377 e2[1]/= e2Norm;
378 e2[2]/= e2Norm;
379
380 // e1 = e2 x e3
381 point_type e1;
382 e1[0]= e2[1] * e3[2] - e2[2] * e3[1];
383 e1[1]= e2[2] * e3[0] - e2[0] * e3[2];
384 e1[2]= e2[0] * e3[1] - e2[1] * e3[0];
385
386 // Transform local coordinates to Cartesian coordinates
387 point_type point;
388 point[0]= local[0] * e1[0] + local[1] * e2[0] + local[2] * e3[0];
389 point[1]= local[0] * e1[1] + local[1] * e2[1] + local[2] * e3[1];
390 point[2]= local[0] * e1[2] + local[1] * e2[2] + local[2] * e3[2];
391
392 return point;
393 }

422 static bool withinNormalizedVelocityCap(const point_type &point, RealType theta_perp,

425 RealType norm_x= std::abs(point[0]) / theta_perp;
