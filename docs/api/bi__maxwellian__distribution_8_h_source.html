174 template <typename GeneratorType>
175 result_type operator()(GeneratorType &gen, const param_type &params)
176 {
177 const RealType sigmaScale= RealType(1.0) / std::sqrt(RealType(2.0));
178 const int kMaxCapRejectTries= 1000000;
179 RealType maxNormalizedVelocity= params.max_normalized_velocity();
180
181 int tries= 0;
182 for (tries= 0; tries < kMaxCapRejectTries; ++tries)
183 {
184 RealType vx= params.theta_perp() * sigmaScale * m_standardNormal(gen);
185 RealType vy= params.theta_perp() * sigmaScale * m_standardNormal(gen);
186 RealType vz= params.theta_par() * sigmaScale * m_standardNormal(gen);
187
188 point_type point;
189 point[0]= vx;
190 point[1]= vy;
191 point[2]= vz;
192
193 if (!withinNormalizedVelocityCap(point, params.theta_perp(), params.theta_par(),
194 maxNormalizedVelocity))
195 {
196 continue;
197 }
198
199 // Rotate only when ub is explicitly set (non-zero).
200 point_type ub= params.ub();
201 if (approximatelyEqual(ub[0], RealType(0.0)) &&
202 approximatelyEqual(ub[1], RealType(0.0)) &&
203 approximatelyEqual(ub[2], RealType(1.0)))
204 {
205 return point;
206 }
207 return rotate_from_fieldAligned_frame(point, ub);
208 }
209
210 throw std::runtime_error("failed to sample velocity under max_normalized_velocity cap: "
211 "rejection sampling exceeded internal attempt limit");
212 }

214 // ====================================================================
215 // Geometry helper
216 // ====================================================================
217 // Transform local velocity components to global Cartesian coordinates given the parallel
218 // direction ub.
219 point_type rotate_from_fieldAligned_frame(const point_type &local, const point_type &ub) const
220 {
221 const RealType kZero= RealType(0.0);
222 const RealType kOne= RealType(1.0);
223
224 RealType ubNorm2= ub[0] * ub[0] + ub[1] * ub[1] + ub[2] * ub[2];
225 if (approximatelyEqual(ubNorm2, kZero))
226 {
227 throw std::invalid_argument("ub must be a non-zero vector");
228 }
229
230 RealType ubNorm= std::sqrt(ubNorm2);
231
232 point_type e3;
233 e3[0]= ub[0] / ubNorm;
234 e3[1]= ub[1] / ubNorm;
235 e3[2]= ub[2] / ubNorm;
236
237 point_type e2({kOne, kOne, kOne});
238
239 int maxcomp= 0;
240 if (std::abs(e3[1]) > std::abs(e3[maxcomp]))
241 {
242 maxcomp= 1;
243 }
244 if (std::abs(e3[2]) > std::abs(e3[maxcomp]))
245 {
246 maxcomp= 2;
247 }
248
249 RealType sum= e3[0] + e3[1] + e3[2];
250 e2[maxcomp]= kOne - sum / e3[maxcomp];
251
252 RealType e2Norm2= e2[0] * e2[0] + e2[1] * e2[1] + e2[2] * e2[2];
253 RealType e2Norm= std::sqrt(e2Norm2);
254 e2[0]/= e2Norm;
255 e2[1]/= e2Norm;
256 e2[2]/= e2Norm;
257
258 point_type e1;
259 e1[0]= e2[1] * e3[2] - e2[2] * e3[1];
260 e1[1]= e2[2] * e3[0] - e2[0] * e3[2];
261 e1[2]= e2[0] * e3[1] - e2[1] * e3[0];
262
263 point_type point;
264 point[0]= local[0] * e1[0] + local[1] * e2[0] + local[2] * e3[0];
265 point[1]= local[0] * e1[1] + local[1] * e2[1] + local[2] * e3[1];
266 point[2]= local[0] * e1[2] + local[1] * e2[2] + local[2] * e3[2];
267 return point;
268 }
