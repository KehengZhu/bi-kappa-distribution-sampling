250 template <typename GeneratorType>
251 result_type operator()(GeneratorType &gen, const param_type &params)
252 {
253 const RealType sigmaScale= RealType(1.0) / std::sqrt(RealType(2.0));
254 const int kMaxCapRejectTries= 1000000;
255 RealType maxNormalizedVelocity= params.max_normalized_velocity();
256
257 // Untruncated mode runs the core draw exactly once; capped mode redraws
258 // until the sample falls inside the box.
259 const bool capped= params.capped();
260 const int maxTries= capped ? kMaxCapRejectTries : 1;
261
262 int tries= 0;
263 for (tries= 0; tries < maxTries; ++tries)
264 {
265 RealType vx= params.theta_perp() * sigmaScale * m_standardNormal(gen);
266 RealType vy= params.theta_perp() * sigmaScale * m_standardNormal(gen);
267 RealType vz= params.theta_par() * sigmaScale * m_standardNormal(gen);
268
269 point_type point;
270 point[0]= vx;
271 point[1]= vy;
272 point[2]= vz;
273
274 if (capped && !withinNormalizedVelocityCap(point, params.theta_perp(),
275 params.theta_par(),
276 maxNormalizedVelocity))
277 {
278 continue;
279 }
280
281 // {0,0,1} is the identity rotation, so skip the transform for it.
282 point_type ub= params.ub();
283 if (approximatelyEqual(ub[0], RealType(0.0)) &&
284 approximatelyEqual(ub[1], RealType(0.0)) &&
285 approximatelyEqual(ub[2], RealType(1.0)))
286 {
287 return point;
288 }
289 return rotate_from_fieldAligned_frame(point, ub);
290 }
291
292 throw std::runtime_error("failed to sample velocity under max_normalized_velocity cap: "
293 "rejection sampling exceeded internal attempt limit");
294 }

296 // ====================================================================
297 // Geometry helper
298 // ====================================================================
299 // Transform local velocity components to global Cartesian coordinates given the parallel
300 // direction ub.
301 point_type rotate_from_fieldAligned_frame(const point_type &local, const point_type &ub) const
302 {
303 const RealType kZero= RealType(0.0);
304 const RealType kOne= RealType(1.0);
305
306 RealType ubNorm2= ub[0] * ub[0] + ub[1] * ub[1] + ub[2] * ub[2];
307 if (approximatelyEqual(ubNorm2, kZero))
308 {
309 throw std::invalid_argument("ub must be a non-zero vector");
310 }
311
312 RealType ubNorm= std::sqrt(ubNorm2);
313
314 point_type e3;
315 e3[0]= ub[0] / ubNorm;
316 e3[1]= ub[1] / ubNorm;
317 e3[2]= ub[2] / ubNorm;
318
319 point_type e2({kOne, kOne, kOne});
320
321 int maxcomp= 0;
322 if (std::abs(e3[1]) > std::abs(e3[maxcomp]))
323 {
324 maxcomp= 1;
325 }
326 if (std::abs(e3[2]) > std::abs(e3[maxcomp]))
327 {
328 maxcomp= 2;
329 }
330
331 RealType sum= e3[0] + e3[1] + e3[2];
332 e2[maxcomp]= kOne - sum / e3[maxcomp];
333
334 RealType e2Norm2= e2[0] * e2[0] + e2[1] * e2[1] + e2[2] * e2[2];
335 RealType e2Norm= std::sqrt(e2Norm2);
336 e2[0]/= e2Norm;
337 e2[1]/= e2Norm;
338 e2[2]/= e2Norm;
339
340 point_type e1;
341 e1[0]= e2[1] * e3[2] - e2[2] * e3[1];
342 e1[1]= e2[2] * e3[0] - e2[0] * e3[2];
343 e1[2]= e2[0] * e3[1] - e2[1] * e3[0];
344
345 point_type point;
346 point[0]= local[0] * e1[0] + local[1] * e2[0] + local[2] * e3[0];
347 point[1]= local[0] * e1[1] + local[1] * e2[1] + local[2] * e3[1];
348 point[2]= local[0] * e1[2] + local[1] * e2[2] + local[2] * e3[2];
349 return point;
350 }
