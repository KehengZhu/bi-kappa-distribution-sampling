1"""Low-kappa numerical-risk audit for the bi-Kappa Gamma-ratio radial draw.
2
3Question under test
4-------------------
5Does the bounded transform Y = T/(1+T) ~ Beta(3/2, kappa-1/2) remove the
6zero-division failure mode that Zenitani et al. (2026) flag for
7shape(X2) = kappa - 1/2 < 1, or does it only relocate it?
8
9Answer: it relocates it AND makes it far worse. See main() output.
10
11Radial construction under test (cpp/bi_kappa_distribution.H:239):
12 X1 ~ Ga(3/2, 1), X2 ~ Ga(kappa - 1/2, 1), R = sqrt(X1 / X2)
13
14Variants compared
15 naive : R = sqrt(X1 / X2) <- what the C++ does today
16 split : R = sqrt(X1) / sqrt(X2) <- avoids the intermediate T overflow
17 logdom : R = exp(0.5*(log X1 - log X2)) <- log of the *already drawn* X2
18 beta : Y = X1/(X1+X2), T = Y/(1-Y), R = sqrt(T) <- the proposed "fix"
19 logga : log-space gamma, log X2 never materialised as a float
20
21Run: python3 python/lowkappa_risk_audit.py
22"""

32def draw(kappa, n, rng):
33 x1 = rng.standard_gamma(1.5, n)
34 x2 = rng.standard_gamma(kappa - 0.5, n)
35 return x1, x2

38def log_gamma_small_shape(b, n, rng):
39 """log of a Ga(b,1) variate for b < 1, computed without underflow.
40
41 numpy and libstdc++ both realise a small-shape gamma via the Marsaglia-Tsang
42 boost X = Ga(b+1) * U**(1/b). Taking the log of that identity,
43 log X = log Ga(b+1) + (1/b) * log U
44 is exactly computable: log U is O(-1e2) and (1/b)*log U is O(-1e5), both far
45 inside double range, even where X itself flushes to zero.
46 """
47 g = rng.standard_gamma(b + 1.0, n)
48 u = rng.uniform(size=n)
49 return np.log(g) + np.log(u) / b

52def variants(x1, x2):
53 with np.errstate(divide="ignore", invalid="ignore", over="ignore"):
54 naive = np.sqrt(x1 / x2)
55 split = np.sqrt(x1) / np.sqrt(x2)
56 logdom = np.exp(0.5 * (np.log(x1) - np.log(x2)))
57 y = x1 / (x1 + x2)
58 beta = np.sqrt(y / (1.0 - y))
59 return {"naive": naive, "split": split, "logdom": logdom, "beta": beta}
