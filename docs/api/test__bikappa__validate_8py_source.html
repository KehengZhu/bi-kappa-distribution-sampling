26def draw(n, kappa, theta_perp, theta_par, rng, uniform_theta=False, bhat=None):
27 """A reference bi-Kappa sample, independent of the C++ implementation."""
28 # sqrt(X1)/sqrt(X2), never sqrt(X1/X2): the quotient overflows at small
29 # kappa in a regime where the radius itself is perfectly representable.
30 R = np.sqrt(rng.gamma(1.5, 1.0, n)) / np.sqrt(rng.gamma(kappa - 0.5, 1.0, n))
31 if uniform_theta: # the classic direction bug
32 c = np.cos(rng.uniform(0.0, np.pi, n))
33 else:
34 c = rng.uniform(-1.0, 1.0, n)
35 phi = rng.uniform(0.0, 2.0 * np.pi, n)
36 s = np.sqrt(1.0 - c * c)
37 u = np.column_stack([R * s * np.cos(phi), R * s * np.sin(phi), R * c])
38 v = u * (np.sqrt(kappa) * np.array([theta_perp, theta_perp, theta_par]))
39 if bhat is not None:
40 v = v @ field_basis(bhat).T
41 return v

44def failing_tests(report):
45 bad = [t for t, d in report["family"]["tests"].items() if d["rejected_at_alpha"]]
46 if not report["tests"]["anisotropy"]["passed"]:
47 bad.append("anisotropy")
48 if not report["tests"]["finiteness"]["passed"]:
49 bad.append("finiteness")

102 # A radius-direction coupling that leaves both marginals correct: rotate the
103 # direction of the outer half of the sample onto a preferred axis by pairing
104 # sorted radii with sorted |cos Theta|. Marginals are untouched by
105 # construction; only the joint law is wrong.
106 v = draw(N, 2.0, 1.0, 1.0, rng)
107 R = np.hypot(np.hypot(v[:, 0], v[:, 1]), v[:, 2])
108 order_r = np.argsort(R)
109 c = v[:, 2] / R
110 order_c = np.argsort(np.abs(c))
111 v2 = np.empty_like(v)
112 v2[order_r] = (v[order_c] / np.linalg.norm(v[order_c], axis=1)[:, None]) * R[order_r, None]
113 r = validate_sample(v2, 2.0, 1.0, 1.0)
114 check("radius-direction coupling is caught with marginals intact",
115 "independence_chi2" in failing_tests(r) or "independence_inner_outer" in failing_tests(r),
116 f"caught by: {failing_tests(r)}")

118 print("bound accounting")

120 v = draw(4 * N, 2.0, 1.0, 2.0, rng)

122 vc, attempts = v[keep], v.shape[0]
