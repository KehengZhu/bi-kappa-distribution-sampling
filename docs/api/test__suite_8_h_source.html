38// ---- polygamma -------------------------------------------------------------
39// psi^(m)(z) by upward recurrence to z >= 20 followed by the Stirling-type
40// asymptotic series. Only m = 0, 1, 3 are needed: E[log X] = psi(a),
41// Var[log X] = psi'(a), and the standard error of a sample variance needs the
42// fourth central moment mu4 = psi'''(a) + 3 psi'(a)^2.

44static double test_digamma(double z)
45{
46 double r= 0.0;
47 while (z < 20.0)
48 {
49 r-= 1.0 / z;
50 z+= 1.0;
51 }
52 const double w= 1.0 / (z * z);
53 return r + std::log(z) - 0.5 / z -
54 w * (1.0 / 12.0 -
55 w * (1.0 / 120.0 - w * (1.0 / 252.0 - w * (1.0 / 240.0 - w * (1.0 / 132.0)))));
56}

101static double test_binom_log_pmf(long long k, long long n, double p)
102{
103 if (k < 0 || k > n)
104 return -std::numeric_limits<double>::infinity();
105 const double dk= static_cast<double>(k), dn= static_cast<double>(n);
106 double t= std::lgamma(dn + 1.0) - std::lgamma(dk + 1.0) - std::lgamma(dn - dk + 1.0);
107 if (k > 0)
108 t+= dk * std::log(p);
109 if (k < n)
110 t+= (dn - dk) * std::log1p(-p);
111 return t;
112}

118static void test_binom_tails(long long k, long long n, double p, double *at_most,
119 double *at_least)
120{
121 const double mean= static_cast<double>(n) * p;
122 const double sd= std::sqrt(static_cast<double>(n) * p * (1.0 - p));
123 long long lo= static_cast<long long>(std::floor(mean - 40.0 * sd - 80.0));
124 long long hi= static_cast<long long>(std::ceil(mean + 40.0 * sd + 80.0));
125 if (lo < 0)
126 lo= 0;
127 if (hi > n)
128 hi= n;
129 double below= 0.0, above= 0.0, at= 0.0;
130 for (long long i= lo; i <= hi; ++i)
131 {
132 const double w= std::exp(test_binom_log_pmf(i, n, p));
133 if (i < k)
134 below+= w;
135 else if (i > k)
136 above+= w;
137 else
138 at= w;
139 }
140 const double total= below + at + above;
141 *at_most= (total > 0.0) ? (below + at) / total : 0.0;
142 *at_least= (total > 0.0) ? (above + at) / total : 0.0;
143}

155// ---- the analytic representability floor -----------------------------------
156//
157// With a = kappa - 1/2, X1 ~ Gamma(3/2,1), Y ~ Gamma(a+1,1), U ~ Uniform(0,1) and
158// E = -log U ~ Exp(1), the shape-boosting identity gives log X2 = log Y - E/a, so
159//
160// log R = (log X1 - log Y + E/a) / 2
161//
162// and therefore, for any threshold L,
163//
164// P(log R > L) = P(E > 2aL - a log(X1/Y)) = E[(X1/Y)^a] exp(-2aL),
165//
166// because E is exponential and independent of X1 and Y. The prefactor is exact:
167// E[X1^a] = Gamma(3/2+a)/Gamma(3/2) and E[Y^-a] = 1/Gamma(a+1).
168//
169// A returned component is V_j = R g_j with g = Q(ub) diag(sqrt(kappa) theta) n and
170// n uniform on the sphere, so the draw has no representation exactly when
171// R max_j |g_j| exceeds the largest finite value MAX of the type:
172//
173// P(overflow) = [Gamma(3/2+a) / (Gamma(3/2) Gamma(a+1))] MAX^(-2a) E[(max_j |g_j|)^(2a)]
174//
175// The last factor depends only on the direction and is evaluated here by product
176// midpoint quadrature over (cos theta, phi); it converges to six digits well below
177// the grid used.
178
179static double test_radius_tail_prefactor(double a)
180{
181 return std::exp(std::lgamma(1.5 + a) - std::lgamma(1.5) - std::lgamma(1.0 + a));
182}

194static void test_overflow_floor(double kappa, double theta_perp, double theta_par,
195 const std::array<double, 3> &ub, double log_max,
196 double *p_overflow, double *p_rescued)
197{
198 const int n_mu= 512, n_phi= 1024;
199 const double a= kappa - 0.5;
200 const double sqrt_kappa= std::sqrt(kappa);
201 const bool axis_aligned= (ub[0] == 0.0 && ub[1] == 0.0 && ub[2] == 1.0);
202 bi_kappa_distribution<double> rot(2.0, 1.0, 1.0);
203
204 double acc_overflow= 0.0, acc_rescued= 0.0;
205 for (int i= 0; i < n_mu; ++i)
206 {
207 const double mu= -1.0 + (static_cast<double>(i) + 0.5) * 2.0 / n_mu;
208 const double sin_theta= std::sqrt(std::max(0.0, 1.0 - mu * mu));
209 for (int k= 0; k < n_phi; ++k)
210 {
211 const double phi= (static_cast<double>(k) + 0.5) * 2.0 * test_pi() / n_phi;
212 std::array<double, 3> local;
213 local[0]= sqrt_kappa * theta_perp * (sin_theta * std::cos(phi));
214 local[1]= sqrt_kappa * theta_perp * (sin_theta * std::sin(phi));
215 local[2]= sqrt_kappa * theta_par * mu;
216 const std::array<double, 3> g=
217 axis_aligned ? local : rot.rotate_from_fieldAligned_frame(local, ub);
218 const double m_local=
219 std::max(std::abs(local[0]), std::max(std::abs(local[1]), std::abs(local[2])));
220 const double m_rot= std::max(std::abs(g[0]), std::max(std::abs(g[1]), std::abs(g[2])));
221 const double e_rot= std::exp(2.0 * a * (std::log(m_rot) - log_max));
222 acc_overflow+= e_rot;
223 const double gap= std::exp(2.0 * a * (std::log(m_local) - log_max)) - e_rot;
224 if (gap > 0.0)
225 acc_rescued+= gap;
226 }
227 }
228 const double cells= static_cast<double>(n_mu) * static_cast<double>(n_phi);
229 const double c= test_radius_tail_prefactor(a);
230 *p_overflow= c * acc_overflow / cells;
231 *p_rescued= c * acc_rescued / cells;
232}

268// ---- replay of the sampler's primitive sequence ----------------------------
269//
270// The sampler draws, in this order: X1 ~ Gamma(3/2,1); log X2 by the boosting
271// identity at shape a; then a direction by Marsaglia's rejection method, two
272// open-interval uniforms per attempt, averaging 4/pi attempts. The second
273// polar-normal variate is cached across calls, so a replay has to carry the same
274// cache or it desynchronizes after the first draw; the direction's rejection loop
275// makes the per-draw variate count variable, so a replay assuming a fixed count
276// would desynchronize too. Replaying lets a test see the quantities the sampler
277// decides on -- log R, the direction, the pre-rotation vector -- which are
278// otherwise private.
279
280struct test_replay_state
281{
282 double cache;
283 bool cached;
284 test_replay_state() : cache(0.0), cached(false) {}
285};

289 double log_r;
290 double nx, ny, nz;
291};

293template <typename GeneratorType>
294static test_replay_draw test_replay(double a, GeneratorType &gen, test_replay_state &st)
295{
296 const double x1= bikappa_detail::marsagliaTsangGamma<double, GeneratorType>(
297 1.5, gen, st.cache, st.cached);
298 const double log_x2=
299 bikappa_detail::logGamma<double, GeneratorType>(a, gen, st.cache, st.cached);
300 test_replay_draw d;
301 d.log_r= 0.5 * (std::log(x1) - log_x2);
302 double d1, d2, dsq;
303 do
304 {
305 d1= 2.0 * bikappa_detail::openCanonical<double, GeneratorType>(gen) - 1.0;
306 d2= 2.0 * bikappa_detail::openCanonical<double, GeneratorType>(gen) - 1.0;
307 dsq= d1 * d1 + d2 * d2;
308 } while (dsq >= 1.0);
309 const double dscale= 2.0 * std::sqrt(1.0 - dsq);
310 d.nx= d1 * dscale;
311 d.ny= d2 * dscale;
312 d.nz= 1.0 - 2.0 * dsq;
313 return d;
314}

1459// ================================================================
1460// test_radius_formation
1461//
1462// Regression tests for the radial formation in bi_kappa_distribution.
1463//
1464// The radius is R = sqrt(X1/X2) with X1 ~ Ga(3/2,1) and X2 ~ Ga(kappa-1/2,1).
1465// The two square roots must be taken SEPARATELY:
1466//
1467// sqrt(x1) / sqrt(x2) not sqrt(x1 / x2)
1468//
1469// The two are algebraically identical but not numerically identical. For small
1470// kappa the shape of X2 is far below 1, X2 is frequently driven into the
1471// subnormal range, and the intermediate quotient x1/x2 overflows to +inf even
1472// where the final radius R is comfortably representable. Taking the roots first
1473// halves the exponent before the division and keeps those draws.
1474//
1475// Measured effect (experiments/exp4_precision, 5e6 draws per configuration,
1476// identical variates through both formations, libc++ and libstdc++ agreeing to
1477// within seed noise):
1478//
1479// float, kappa = 0.55 loss 1.22e-2 -> 5.7e-3 (fix recovers 6.5e-3)
1480// float, kappa = 0.60 loss 1.5e-4 -> 2.9e-5 (about 5x)
1481// double, kappa = 0.51 loss 8.2e-4 -> 5.7e-4
1482//
1483// A1 is a direct, deterministic check of the overflow mechanism itself; A2 and A3
1484// guard the released sampler at the two ends of the range.
1485//
1486// Since 2.0.0 the sampler does not form X2 at all -- it carries log X2 and log R --
1487// so A1 and A4 now describe a formation the released class no longer uses, and what
1488// A2 and A3 measure is the representability floor of the type rather than the
1489// avoidable loss of an intermediate. test_log_radius_finiteness below covers the
1490// log-domain sampler against that floor directly.
1491// ================================================================
1492static int test_radius_formation()
1493{
1494 bool all_ok= true;
1495 std::cout << "\n--- test_radius_formation ---" << std::endl;
1496
1497 // ---- A1: the mechanism, deterministically ----------------------------
1498 // Pick x2 subnormal enough that x1/x2 overflows in float while
1499 // sqrt(x1)/sqrt(x2) does not. No RNG involved.
1500 {
1501 const float x1= 1.0f;
1502 const float x2= std::numeric_limits<float>::denorm_min(); // ~1.4e-45
1503 const float r_ratio= std::sqrt(x1 / x2);
1504 const float r_split= std::sqrt(x1) / std::sqrt(x2);
1505
1506 const bool ratio_overflows= !std::isfinite(r_ratio);

1508 const bool ok= ratio_overflows && split_survives;
1509 all_ok= all_ok && ok;

1511 << " (ratio=" << r_ratio << ", split=" << r_split << "): "
1512 << (ok ? "PASS" : "FAIL") << std::endl;

1515 // ---- A2: the released sampler is clean in double at kappa >= 0.55 ----
1516 // Experiment 4 measures exactly zero non-finite draws over 5e6 draws per

2707// ================================================================
2708// test_log_cap_semantics
2709//
2710// The cap is now decided in the log domain, before anything is exponentiated, so
2711// that a sample far outside the box is rejected rather than overflowed first and
2712// rejected afterwards. That is only legitimate if it accepts exactly the set the
2713// published linear definition accepts.
2714// ================================================================
2715static int test_log_cap_semantics()
2716{
2717 bool all_ok= true;
2718 std::cout << "\n--- test_log_cap_semantics ---" << std::endl;
2719
2720 // ---- Q1: the two predicates accept exactly the same set -----------------
2721 // The linear predicate |point_j| / theta_j <= lambda is private, so it is
2722 // transcribed here. Note that theta cancels out of it: with
2723 // point = R sqrt(kappa) (theta_perp n_x, theta_perp n_y, theta_par n_z) the
2724 // normalized components are sqrt(kappa) R |n_j|, which is what the log form
2725 // tests. ub is the axis, so the returned vector is the field-aligned one and
2726 // the linear predicate applies to it unchanged.
2727 //
2728 // The replayed accepted draws are also compared against the sampler's own
2729 // returned stream bitwise, so this checks the transcription as well as the
2730 // predicates.
2731 {
2732 const long long attempts= 300000;

2734 const double a= kappa - 0.5, sqrt_kappa= std::sqrt(kappa);
2735 const double log_cap= std::log(lambda), log_sqrt_kappa= std::log(sqrt_kappa);

2737 bi_kappa_distribution<double> dist(kappa, theta_perp, theta_par, {0.0, 0.0, 1.0},

2740 test_replay_state st;

2742 for (long long i= 0; i < attempts; ++i)

2745 const double max_abs_n=
2746 std::max(std::abs(d.nx), std::max(std::abs(d.ny), std::abs(d.nz)));
2747 const double log_largest= d.log_r + log_sqrt_kappa + std::log(max_abs_n);
2748 const bool log_accepts= !(log_largest > log_cap);
2749
2750 const double r= std::exp(d.log_r);
2751 bi_kappa_distribution<double>::point_type point;
2752 point[0]= r * (sqrt_kappa * theta_perp * d.nx);
2753 point[1]= r * (sqrt_kappa * theta_perp * d.ny);
2754 point[2]= r * (sqrt_kappa * theta_par * d.nz);
2755 if (!std::isfinite(r) || !std::isfinite(point[0]) || !std::isfinite(point[1]) ||
2756 !std::isfinite(point[2]))
2757 continue; // not representable in the linear form; nothing to compare

2761 (std::abs(point[1]) / theta_perp <= lambda) &&

2830// ================================================================
2831// test_log_rotation_stability
2832//
2833// An orthogonal rotation can spread one over-large field-aligned component across
2834// three representable global ones, which is why the sampler builds the order-unity
2835// vector g = Q(ub) diag(sqrt(kappa) theta) n first and decides representability from
2836// log|V_j| = log R + log|g_j| on the components it is actually going to return.
2837// Deciding on the pre-rotation vector instead would discard draws that have a
2838// perfectly good answer.
2839// ================================================================
2840static int test_log_rotation_stability()
2841{
2842 bool all_ok= true;
2843 std::cout << "\n--- test_log_rotation_stability ---" << std::endl;
2844
2845 // ---- R1: finiteness follows the log-domain predicate, and the rotation is
2846 // accurate -----------------------------------------------------
2847 // The reference recomputes the same draw in long double the other way round --
2848 // scale first, then rotate -- so the comparison is between two different
2849 // orderings of the same algebra rather than a repetition of one. A rotation is
2850 // backward stable in the norm but not component by component: with ub = (1,1,1)
2851 // the three products cancel, so a component that is small compared with the
2852 // vector carries a large relative error and a small absolute one. The tolerance
2853 // is therefore stated in ulp of the vector's largest component, which is the
2854 // quantity a rotation error is actually bounded by.
2855 //
2856 // Note that long double is not wider than double on every platform; this build
2857 // reports its mantissa below, and where the two coincide the reference is an
2858 // independent recomputation at the same precision rather than a more accurate
2859 // one.
2860 {
2861 const long long n= 200000;
2862 const double kappa= 0.51, theta_perp= 1.0, theta_par= 2.0;
2863 const double a= kappa - 0.5, sqrt_kappa= std::sqrt(kappa);
2864 const double log_max= std::log(std::numeric_limits<double>::max());
2865 const bi_kappa_distribution<double>::point_type ub= {1.0, 1.0, 1.0};
2866 const std::array<long double, 3> ub_ld= {1.0L, 1.0L, 1.0L};
2867
2868 bi_kappa_distribution<double> dist(kappa, theta_perp, theta_par, ub);
2869 bi_kappa_distribution<long double> rot_ld(2.0L, 1.0L, 1.0L);
2870 std::mt19937 gen_sampler(90070u), gen_replay(90070u);
2871 test_replay_state st;
2872
2873 long long predicate_mismatch= 0, compared= 0, skipped= 0;
2874 double worst_ulp= 0.0;
2875 for (long long i= 0; i < n; ++i)
2876 {
2877 const test_replay_draw d= test_replay(a, gen_replay, st);
2878 bi_kappa_distribution<double>::point_type local;
2879 local[0]= sqrt_kappa * theta_perp * d.nx;
2880 local[1]= sqrt_kappa * theta_perp * d.ny;
2881 local[2]= sqrt_kappa * theta_par * d.nz;
2882 const bi_kappa_distribution<double>::point_type g=
2883 dist.rotate_from_fieldAligned_frame(local, ub);
2884
2885 bool predicted_finite= true;
2886 for (int j= 0; j < 3; ++j)
2887 {
2888 if (g[j] != 0.0 && d.log_r + std::log(std::abs(g[j])) >= log_max)
2889 predicted_finite= false;
2890 }
2891
2892 const bi_kappa_distribution<double>::result_type v= dist(gen_sampler);
2893 const bool actually_finite=
2894 std::isfinite(v[0]) && std::isfinite(v[1]) && std::isfinite(v[2]);
2895 if (predicted_finite != actually_finite)
2896 ++predicate_mismatch;
2897 if (!actually_finite)
2898 continue;
2899
2900 const long double r_ld= std::exp(static_cast<long double>(d.log_r));
2901 std::array<long double, 3> scaled;
2902 bool scaled_ok= true;
2903 for (int j= 0; j < 3; ++j)
2904 {
2905 scaled[j]= r_ld * static_cast<long double>(local[j]);
2906 if (!std::isfinite(static_cast<double>(scaled[j])))
2907 scaled_ok= false;
2908 }
2909 if (!scaled_ok)
2910 {
2911 // rotation-recoverable draw: the pre-rotation vector has no double
2912 // representation, so scale-then-rotate has nothing to compare.
2913 ++skipped;
2914 continue;
2915 }
2916 const std::array<long double, 3> reference=
2917 rot_ld.rotate_from_fieldAligned_frame(scaled, ub_ld);
2918 const double scale=
2919 std::max(std::abs(v[0]), std::max(std::abs(v[1]), std::abs(v[2])));
2920 ++compared;
2921 for (int j= 0; j < 3; ++j)
2922 {
2923 const double u=
2924 test_ulp_distance(v[j], static_cast<double>(reference[j]), scale);
2925 if (u > worst_ulp)
2926 worst_ulp= u;
2927 }
2928 }
2929 const bool ok= (predicate_mismatch == 0) && (worst_ulp <= 8.0) && (compared > 0);
2930 all_ok= all_ok && ok;
2931 std::cout << " R1 kappa=0.51, ub={1,1,1}: finite exactly where log R + log|g_j| <"
2932 " log DBL_MAX for all j (" << predicate_mismatch
2933 << " mismatches in " << n << "), and within " << worst_ulp
2934 << " ulp of the vector scale against a long double reference with "
2935 << std::numeric_limits<long double>::digits
2936 << "-bit mantissa (bound 8, " << compared << " compared, " << skipped
2937 << " rotation-recoverable): " << (ok ? "PASS" : "FAIL") << std::endl;
2938 }
2939
2940 // ---- R2: the rotation-recoverable case exists and is kept ---------------
2941 // A draw whose field-aligned vector has no representation can still have three
2942 // representable global components, because a rotation can spread it: with
2943 // ub = (1,1,1) a vector along the field maps to (1,1,1)/sqrt(3) times its norm,
2944 // so the largest component falls by sqrt(3). The set is therefore the band of
2945 // log R of width log sqrt(3) just above the threshold, and its probability comes
2946 // from the same closed form as the floor: about 2.4e-6 here, so a few tens of
2947 // millions of attempts are needed before "a few" is more than noise. At 3e7 the
2948 // expected count is about 71, which is far enough from zero that the existence
2949 // claim is not a coin flip and well enough determined that the count itself can
2950 // be tested against the closed form.
2951 {
2952 const long long n= 30000000;
2953 const double kappa= 0.505, theta_perp= 1.0, theta_par= 8.0;
2954 const double a= kappa - 0.5, sqrt_kappa= std::sqrt(kappa);
2955 const double log_max= std::log(std::numeric_limits<double>::max());

2959 test_overflow_floor(kappa, theta_perp, theta_par, ub_q, log_max, &p_overflow,
2960 &p_rescued);
2961
2962 bi_kappa_distribution<double> dist(kappa, theta_perp, theta_par, ub);
2963 std::mt19937 gen_sampler(90075u), gen_replay(90075u);
2964 test_replay_state st;
2965 long long rescued= 0, rescued_lost= 0;
2966 for (long long i= 0; i < n; ++i)
2967 {
2968 const test_replay_draw d= test_replay(a, gen_replay, st);
2969 bi_kappa_distribution<double>::point_type local;
2970 local[0]= sqrt_kappa * theta_perp * d.nx;
2971 local[1]= sqrt_kappa * theta_perp * d.ny;
2972 local[2]= sqrt_kappa * theta_par * d.nz;
2973 const bi_kappa_distribution<double>::point_type g=
2974 dist.rotate_from_fieldAligned_frame(local, ub);
2975 const double m_local= std::max(std::abs(local[0]),
2976 std::max(std::abs(local[1]), std::abs(local[2])));
2977 const double m_rot=
2978 std::max(std::abs(g[0]), std::max(std::abs(g[1]), std::abs(g[2])));
2979 const bool local_overflows= (d.log_r + std::log(m_local) >= log_max);
2980 const bool rotated_fits= (d.log_r + std::log(m_rot) < log_max);
2981
2982 const bi_kappa_distribution<double>::result_type v= dist(gen_sampler);
2983 if (local_overflows && rotated_fits)
2984 {
2985 ++rescued;
2986 if (!std::isfinite(v[0]) || !std::isfinite(v[1]) || !std::isfinite(v[2]))
2987 ++rescued_lost;
2988 }
2989 }
2990 const bool consistent= test_binom_consistent(rescued, n, p_rescued, 1.0e-3);
2991 const bool ok= (rescued > 0) && (rescued_lost == 0) && consistent;
2992 all_ok= all_ok && ok;
2993 std::cout << " R2 kappa=0.505, theta_par/theta_perp=8, ub={1,1,1}: " << rescued
2994 << " rotation-recoverable draws in " << n << " (expected "
2995 << p_rescued * static_cast<double>(n)
2996 << " from the closed form, exact binomial alpha=1e-3), all returned"
2997 " finite (" << rescued_lost << " lost): " << (ok ? "PASS" : "FAIL")
2998 << std::endl;
2999 }
3000
3001 std::cout << "test_log_rotation_stability: " << (all_ok ? "PASS" : "FAIL") << std::endl;
3002 return all_ok ? 0 : 1;
3003}
