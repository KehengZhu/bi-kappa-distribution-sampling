166 template <typename GeneratorType>
167 result_type operator()(GeneratorType &gen, const param_type &params)
168 {
169 const RealType sigmaScale= RealType(1.0) / std::sqrt(RealType(2.0));
170 const int kMaxCapRejectTries= 1000000;
171 RealType maxNormalizedVelocity= params.max_normalized_velocity();
172
173 int tries= 0;
174 for (tries= 0; tries < kMaxCapRejectTries; ++tries)
175 {
176 RealType vx= params.theta_perp() * sigmaScale * m_standardNormal(gen);
177 RealType vy= params.theta_perp() * sigmaScale * m_standardNormal(gen);
178 RealType vz= params.theta_par() * sigmaScale * m_standardNormal(gen);
179
180 point_type point;
181 point[0]= vx;
182 point[1]= vy;
183 point[2]= vz;
184
185 if (!withinNormalizedVelocityCap(point, params.theta_perp(), params.theta_par(),
186 maxNormalizedVelocity))
187 {
188 continue;
189 }
190
191 // Rotate only when ub is explicitly set (non-zero).
192 point_type ub= params.ub();
193 if (approximatelyEqual(ub[0], RealType(0.0)) &&
194 approximatelyEqual(ub[1], RealType(0.0)) &&
195 approximatelyEqual(ub[2], RealType(1.0)))
196 {
197 return point;
198 }
199 return rotate_from_fieldAligned_frame(point, ub);
200 }
201
202 throw std::runtime_error("failed to sample velocity under max_normalized_velocity cap: "
203 "rejection sampling exceeded internal attempt limit");
204 }

206 // ====================================================================
207 // Geometry helper
208 // ====================================================================
209 // Transform local velocity components to global Cartesian coordinates given the parallel
210 // direction ub.
211 point_type rotate_from_fieldAligned_frame(const point_type &local, const point_type &ub) const
212 {
213 const RealType kZero= RealType(0.0);
214 const RealType kOne= RealType(1.0);
215
216 RealType ubNorm2= ub[0] * ub[0] + ub[1] * ub[1] + ub[2] * ub[2];
217 if (approximatelyEqual(ubNorm2, kZero))
218 {
219 throw std::invalid_argument("ub must be a non-zero vector");
220 }
221
222 RealType ubNorm= std::sqrt(ubNorm2);
223
224 point_type e3;
225 e3[0]= ub[0] / ubNorm;
226 e3[1]= ub[1] / ubNorm;
227 e3[2]= ub[2] / ubNorm;
228
229 point_type e2({kOne, kOne, kOne});
230
231 int maxcomp= 0;
232 if (std::abs(e3[1]) > std::abs(e3[maxcomp]))
233 {
234 maxcomp= 1;
235 }
236 if (std::abs(e3[2]) > std::abs(e3[maxcomp]))
237 {
238 maxcomp= 2;
239 }
240
241 RealType sum= e3[0] + e3[1] + e3[2];
242 e2[maxcomp]= kOne - sum / e3[maxcomp];
243
244 RealType e2Norm2= e2[0] * e2[0] + e2[1] * e2[1] + e2[2] * e2[2];
245 RealType e2Norm= std::sqrt(e2Norm2);
246 e2[0]/= e2Norm;
247 e2[1]/= e2Norm;
248 e2[2]/= e2Norm;
249
250 point_type e1;
251 e1[0]= e2[1] * e3[2] - e2[2] * e3[1];
252 e1[1]= e2[2] * e3[0] - e2[0] * e3[2];
253 e1[2]= e2[0] * e3[1] - e2[1] * e3[0];
254
255 point_type point;
256 point[0]= local[0] * e1[0] + local[1] * e2[0] + local[2] * e3[0];
257 point[1]= local[0] * e1[1] + local[1] * e2[1] + local[2] * e3[1];
258 point[2]= local[0] * e1[2] + local[1] * e2[2] + local[2] * e3[2];
259 return point;
260 }
